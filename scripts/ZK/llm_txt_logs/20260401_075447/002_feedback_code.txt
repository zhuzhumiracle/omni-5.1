def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    tilt_cos = kwargs["tilt_cos"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    progress = prev_dist - curr_dist
    progress_reward = 12.0 * torch.tanh(4.0 * progress)

    goal_proximity = torch.exp(-0.8 * curr_dist)
    target_reward = 10.0 * goal_proximity

    success_bonus = (curr_dist < 0.6).float() * 100.0

    safe_lidar = torch.clamp(min_lidar_dist, min=0.0)
    collision_gate = torch.sigmoid((0.35 - safe_lidar) * 35.0)
    severe_collision_penalty = -150.0 * collision_gate
    near_obstacle_penalty = -20.0 * torch.exp(-10.0 * safe_lidar)
    extreme_collision_penalty = -50.0 * torch.exp(-40.0 * safe_lidar)

    speed_penalty = -0.8 * torch.square(torch.relu(v_norm - 4.5))
    low_speed_penalty = -0.2 * torch.relu(0.3 - v_norm)

    omega_norm = torch.norm(omega, dim=-1)
    smooth_penalty = -0.06 * torch.clamp(omega_norm + action_diff, max=100.0)

    altitude_error = torch.abs(z_pos - 2.0)
    altitude_penalty = -1.5 * torch.square(torch.relu(altitude_error - 0.25))

    tilt_penalty = -1.0 * torch.relu(0.85 - torch.clamp(tilt_cos, min=-1.0, max=1.0))

    to_target = target_pos - curr_pos
    to_target_norm = torch.norm(to_target, dim=-1).clamp_min(1e-6)
    to_target_dir = to_target / to_target_norm.unsqueeze(-1)

    x_body_norm = torch.norm(x_body, dim=-1).clamp_min(1e-6)
    x_body_dir = x_body / x_body_norm.unsqueeze(-1)
    align_reward = 1.5 * torch.sum(x_body_dir * to_target_dir, dim=-1)

    v_norm_safe = v_norm.clamp_min(1e-6)
    velocity_dir = v / v_norm_safe.unsqueeze(-1)
    move_align = 0.8 * torch.sum(velocity_dir * to_target_dir, dim=-1) * torch.tanh(v_norm)

    shaping = 0.5 * torch.exp(-0.15 * curr_dist)

    reward = (
        progress_reward
        + target_reward
        + success_bonus
        + severe_collision_penalty
        + near_obstacle_penalty
        + extreme_collision_penalty
        + speed_penalty
        + low_speed_penalty
        + smooth_penalty
        + altitude_penalty
        + tilt_penalty
        + align_reward
        + move_align
        + shaping
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    reward = reward.reshape(-1)
    return reward
