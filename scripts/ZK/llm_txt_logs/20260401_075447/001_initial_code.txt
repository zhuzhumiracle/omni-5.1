def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    tilt_cos = kwargs["tilt_cos"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    progress = prev_dist - curr_dist
    progress_reward = torch.tanh(3.0 * progress)

    dist_to_goal = torch.clamp(curr_dist, min=0.0)
    reach_reward = 8.0 * torch.exp(-2.0 * dist_to_goal)

    z_error = z_pos - 2.0
    altitude_penalty = torch.clamp(torch.abs(z_error) - 0.25, min=0.0)
    altitude_reward = -1.5 * torch.square(altitude_penalty)

    safe_lidar = torch.clamp(min_lidar_dist, min=0.0)
    obstacle_penalty = -3.0 * torch.exp(-6.0 * safe_lidar)
    severe_collision_penalty = -10.0 * torch.exp(-20.0 * safe_lidar)

    speed_excess = torch.relu(v_norm - 4.0)
    speed_penalty = -0.6 * torch.square(speed_excess)

    omega_norm = torch.norm(omega, dim=-1)
    smooth_penalty = -0.08 * action_diff - 0.03 * omega_norm

    upright_reward = 1.2 * torch.clamp(tilt_cos, min=-1.0, max=1.0)

    to_target = target_pos - curr_pos
    to_target_norm = torch.norm(to_target, dim=-1).clamp_min(1e-6)
    to_target_dir = to_target / to_target_norm.unsqueeze(-1)
    x_body_norm = torch.norm(x_body, dim=-1).clamp_min(1e-6)
    x_body_dir = x_body / x_body_norm.unsqueeze(-1)
    align_reward = 0.8 * torch.sum(x_body_dir * to_target_dir, dim=-1)

    velocity_dir = v / v_norm.unsqueeze(-1).clamp_min(1e-6)
    move_align = 0.3 * torch.sum(velocity_dir * to_target_dir, dim=-1) * torch.tanh(v_norm)

    close_bonus = 2.5 * (curr_dist < 0.8).float() * torch.exp(-3.0 * curr_dist)

    reward = (
        3.0 * progress_reward
        + reach_reward
        + altitude_reward
        + obstacle_penalty
        + severe_collision_penalty
        + speed_penalty
        + smooth_penalty
        + upright_reward
        + align_reward
        + move_align
        + close_bonus
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    reward = reward.reshape(-1)
    return reward
