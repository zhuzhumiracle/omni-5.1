def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    tilt_cos = kwargs.get("tilt_cos", None)
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    dist_delta = prev_dist - curr_dist
    progress_reward = 18.0 * torch.tanh(3.0 * dist_delta)

    target_reward = 12.0 * torch.exp(-0.35 * curr_dist)
    success_bonus = (curr_dist < 0.7).float() * 150.0

    lidar = min_lidar_dist.clamp_min(0.0)
    obstacle_core = -80.0 * torch.exp(-8.0 * lidar)
    obstacle_hard = -250.0 * torch.exp(-40.0 * lidar)
    collision_zone = -500.0 * torch.sigmoid((0.45 - lidar) * 30.0)
    obstacle_penalty = obstacle_core + obstacle_hard + collision_zone

    speed_penalty = -0.7 * torch.square(torch.relu(v_norm - 5.0))
    low_speed_penalty = -0.15 * torch.relu(0.35 - v_norm)

    omega_norm = torch.norm(omega, dim=-1)
    smooth_penalty = -0.08 * torch.clamp(omega_norm + action_diff, max=50.0)

    altitude_error = torch.abs(z_pos - 2.0)
    altitude_penalty = -1.5 * torch.square(torch.relu(altitude_error - 0.25))

    if tilt_cos is None:
        tilt_penalty = torch.zeros_like(curr_dist)
    else:
        tilt_penalty = -1.0 * torch.relu(0.85 - torch.clamp(tilt_cos, min=-1.0, max=1.0))

    to_target = target_pos - curr_pos
    to_target_norm = torch.norm(to_target, dim=-1).clamp_min(1e-6)
    to_target_dir = to_target / to_target_norm.unsqueeze(-1)

    x_body_norm = torch.norm(x_body, dim=-1).clamp_min(1e-6)
    x_body_dir = x_body / x_body_norm.unsqueeze(-1)
    align_reward = 2.0 * torch.sum(x_body_dir * to_target_dir, dim=-1)

    v_norm_safe = v_norm.clamp_min(1e-6)
    vel_dir = v / v_norm_safe.unsqueeze(-1)
    move_align = 1.0 * torch.sum(vel_dir * to_target_dir, dim=-1) * torch.tanh(v_norm)

    shaping = 1.0 * torch.exp(-0.12 * curr_dist)

    reward = (
        progress_reward
        + target_reward
        + success_bonus
        + obstacle_penalty
        + speed_penalty
        + low_speed_penalty
        + smooth_penalty
        + altitude_penalty
        + tilt_penalty
        + align_reward
        + move_align
        + shaping
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    reward = reward.reshape(-1)
    return reward
