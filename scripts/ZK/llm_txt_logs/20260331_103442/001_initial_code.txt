def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    tilt_cos = kwargs["tilt_cos"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    progress = prev_dist - curr_dist
    progress_reward = 8.0 * torch.tanh(3.0 * progress)

    dist_reward = 2.0 * torch.exp(-0.35 * curr_dist)

    close_to_goal = torch.exp(-0.8 * curr_dist) * 25.0
    goal_bonus = torch.where(curr_dist < 0.75, 150.0 / (curr_dist + 0.05), torch.zeros_like(curr_dist))

    z_err = torch.abs(z_pos - 2.0)
    z_reward = -2.5 * torch.tanh(1.2 * z_err)

    speed_penalty = 0.4 * torch.relu(v_norm - 4.0).pow(2) + 0.05 * v_norm.pow(2)

    omega_norm = torch.norm(omega, dim=-1)
    smooth_penalty = 0.08 * action_diff + 0.03 * omega_norm

    upright_reward = 1.5 * tilt_cos

    alignment = torch.sum(
        x_body * (target_pos - curr_pos) / torch.norm(target_pos - curr_pos, dim=-1, keepdim=True).clamp_min(1e-6),
        dim=-1
    )
    alignment_reward = 0.8 * alignment

    safe_dist = min_lidar_dist.clamp_min(1e-3)
    obstacle_penalty = -3.0 / safe_dist.pow(2) - 0.8 * torch.exp(-4.0 * safe_dist)

    collision_penalty = torch.where(min_lidar_dist < 0.25, -200.0 / safe_dist, torch.zeros_like(min_lidar_dist))

    reward = (
        progress_reward
        + dist_reward
        + close_to_goal
        + goal_bonus
        + z_reward
        - speed_penalty
        - smooth_penalty
        + upright_reward
        + alignment_reward
        + obstacle_penalty
        + collision_penalty
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    return reward
