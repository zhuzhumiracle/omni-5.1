def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    tilt_cos = kwargs["tilt_cos"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    progress = prev_dist - curr_dist
    progress = torch.clamp(progress, min=-2.0, max=2.0)

    target_bonus = 3.5 * torch.exp(-0.7 * curr_dist)
    success_bonus = (curr_dist < 0.75).to(curr_dist.dtype) * 25.0

    altitude_err = torch.abs(z_pos - 2.0)
    altitude_penalty = torch.relu(altitude_err - 0.15)
    altitude_penalty = altitude_penalty * altitude_penalty
    climb_penalty = torch.relu(z_pos - 2.25)
    climb_penalty = climb_penalty * climb_penalty
    low_alt_penalty = torch.relu(1.05 - z_pos)
    low_alt_penalty = low_alt_penalty * low_alt_penalty

    speed_excess = torch.relu(v_norm - 4.0)
    speed_penalty = speed_excess * speed_excess

    omega_mag = torch.norm(omega, dim=-1)
    smooth_penalty = torch.clamp(action_diff + 0.12 * omega_mag, min=0.0, max=25.0)

    v_dir = v / v_norm.unsqueeze(-1).clamp_min(1e-6)
    heading_align = (x_body * v_dir).sum(dim=-1)
    heading_align = torch.clamp(heading_align, min=-1.0, max=1.0)
    heading_reward = torch.where(v_norm > 0.25, heading_align * torch.tanh(v_norm), torch.zeros_like(v_norm))

    upright_reward = torch.clamp(tilt_cos, min=-1.0, max=1.0)

    safe_lidar = torch.clamp(min_lidar_dist, min=1e-6)
    obstacle_penalty = 0.25 / safe_lidar
    obstacle_penalty = torch.clamp(obstacle_penalty, max=12.0)

    near_obstacle_penalty = torch.relu(0.8 - min_lidar_dist)
    near_obstacle_penalty = near_obstacle_penalty * near_obstacle_penalty * 10.0

    ceiling_penalty = torch.relu(z_pos - 2.35)
    ceiling_penalty = ceiling_penalty * ceiling_penalty * 12.0

    reward = (
        6.0 * progress
        + target_bonus
        + success_bonus
        + 1.2 * upright_reward
        + 0.8 * heading_reward
        - 2.8 * altitude_penalty
        - 3.5 * climb_penalty
        - 2.5 * low_alt_penalty
        - 0.5 * speed_penalty
        - 0.18 * smooth_penalty
        - obstacle_penalty
        - near_obstacle_penalty
        - ceiling_penalty
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    return reward
