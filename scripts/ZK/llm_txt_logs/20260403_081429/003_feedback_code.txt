def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    tilt_cos = kwargs["tilt_cos"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    progress = prev_dist - curr_dist
    progress = torch.clamp(progress, min=-1.5, max=1.5)

    goal_proximity = torch.exp(-0.6 * curr_dist)
    success_bonus = (curr_dist < 0.75).to(curr_dist.dtype) * 35.0

    z_error = torch.abs(z_pos - 2.0)
    altitude_penalty = torch.relu(z_error - 0.12)
    altitude_penalty = altitude_penalty * altitude_penalty

    high_alt_penalty = torch.relu(z_pos - 2.15)
    high_alt_penalty = high_alt_penalty * high_alt_penalty

    low_alt_penalty = torch.relu(1.10 - z_pos)
    low_alt_penalty = low_alt_penalty * low_alt_penalty

    climb_runaway_penalty = torch.relu(v[:, 2] if v.ndim == 2 else v)
    climb_runaway_penalty = climb_runaway_penalty * climb_runaway_penalty

    speed_penalty = torch.relu(v_norm - 4.5)
    speed_penalty = speed_penalty * speed_penalty

    omega_mag = torch.norm(omega, dim=-1)
    smooth_penalty = torch.clamp(action_diff + 0.08 * omega_mag, min=0.0, max=20.0)

    v_dir = v / v_norm.unsqueeze(-1).clamp_min(1e-6)
    heading_align = (x_body * v_dir).sum(dim=-1)
    heading_align = torch.clamp(heading_align, min=-1.0, max=1.0)
    heading_reward = torch.where(v_norm > 0.2, heading_align * torch.tanh(v_norm), torch.zeros_like(v_norm))

    upright_reward = torch.clamp(tilt_cos, min=-1.0, max=1.0)

    safe_lidar = min_lidar_dist.clamp_min(1e-3)
    obstacle_penalty = 1.2 / safe_lidar
    obstacle_penalty = torch.clamp(obstacle_penalty, max=18.0)

    near_obstacle_penalty = torch.relu(0.9 - min_lidar_dist)
    near_obstacle_penalty = near_obstacle_penalty * near_obstacle_penalty * 8.0

    collision_penalty = (min_lidar_dist < 0.35).to(curr_dist.dtype) * 20.0

    ceiling_penalty = torch.relu(z_pos - 2.35)
    ceiling_penalty = ceiling_penalty * ceiling_penalty * 15.0

    reward = (
        10.0 * progress
        + 4.0 * goal_proximity
        + success_bonus
        + 1.0 * upright_reward
        + 0.7 * heading_reward
        - 3.5 * altitude_penalty
        - 4.0 * high_alt_penalty
        - 3.0 * low_alt_penalty
        - 1.2 * climb_runaway_penalty
        - 0.45 * speed_penalty
        - 0.12 * smooth_penalty
        - obstacle_penalty
        - near_obstacle_penalty
        - collision_penalty
        - ceiling_penalty
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    return reward
