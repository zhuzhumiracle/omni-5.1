def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    tilt_cos = kwargs["tilt_cos"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    dist_progress = prev_dist - curr_dist
    dist_progress = torch.clamp(dist_progress, min=-5.0, max=5.0)

    reach_bonus = torch.exp(-1.8 * curr_dist) * 12.0

    height_err = torch.abs(z_pos - 2.0)
    height_penalty = torch.clamp(height_err - 0.25, min=0.0)
    height_penalty = height_penalty * height_penalty

    speed_penalty = torch.relu(v_norm - 4.5)
    speed_penalty = speed_penalty * speed_penalty

    smooth_penalty = action_diff + 0.15 * torch.norm(omega, dim=-1)
    smooth_penalty = torch.clamp(smooth_penalty, min=0.0, max=20.0)

    upright_bonus = torch.clamp(tilt_cos, min=-1.0, max=1.0)

    move_dir = v / v_norm.unsqueeze(-1).clamp_min(1e-6)
    align = (x_body * move_dir).sum(dim=-1)
    align = torch.clamp(align, min=-1.0, max=1.0)
    align_bonus = align * torch.tanh(v_norm)

    obstacle_safe = torch.clamp(min_lidar_dist, min=1e-3)
    obstacle_penalty = 0.6 / obstacle_safe
    obstacle_penalty = torch.clamp(obstacle_penalty, max=1000.0)

    near_obstacle_extra = torch.relu(0.6 - min_lidar_dist)
    near_obstacle_extra = near_obstacle_extra * near_obstacle_extra * 8.0

    reward = (
        4.0 * dist_progress
        + reach_bonus
        + 1.2 * upright_bonus
        + 0.8 * align_bonus
        - 2.5 * height_penalty
        - 0.4 * speed_penalty
        - 0.25 * smooth_penalty
        - obstacle_penalty
        - near_obstacle_extra
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    return reward
