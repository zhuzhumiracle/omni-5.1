def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    progress = prev_dist - curr_dist
    progress_reward = torch.tanh(5.0 * progress)

    target_reward = 2.5 * torch.exp(-1.2 * curr_dist)
    success_bonus = (curr_dist < 0.8).to(curr_dist.dtype) * 20.0
    near_goal_bonus = (curr_dist < 2.0).to(curr_dist.dtype) * torch.exp(-0.7 * curr_dist) * 4.0

    speed_penalty = torch.relu(v_norm - 4.5) ** 2
    smooth_penalty = torch.clamp(action_diff + 0.2 * torch.norm(omega, dim=-1), max=50.0)

    altitude_penalty = torch.relu(torch.abs(z_pos - 2.0) - 0.35) ** 2

    obs = min_lidar_dist.clamp_min(0.0)
    obstacle_soft = torch.exp(-4.0 * obs)
    obstacle_mid = torch.exp(-10.0 * obs)
    obstacle_severe = 1.0 / (obs.clamp_min(1e-6) ** 2)
    collision_penalty = (min_lidar_dist < 0.5).to(min_lidar_dist.dtype) * (8.0 + 18.0 * obstacle_soft)
    crash_penalty = (min_lidar_dist < 0.25).to(min_lidar_dist.dtype) * (60.0 + 120.0 * obstacle_mid + 40.0 * obstacle_severe)

    vel_norm_safe = v_norm.clamp_min(1e-6)
    move_dir = v / vel_norm_safe.unsqueeze(-1)
    heading_alignment = (x_body * move_dir).sum(dim=-1)
    heading_alignment = torch.clamp(heading_alignment, -1.0, 1.0)
    motion_bonus = torch.relu(v_norm - 0.3) * heading_alignment

    reward = (
        8.0 * progress_reward
        + 6.0 * target_reward
        + success_bonus
        + near_goal_bonus
        + 0.8 * motion_bonus
        - 0.4 * speed_penalty
        - 0.15 * smooth_penalty
        - 1.5 * altitude_penalty
        - 6.0 * obstacle_soft
        - 12.0 * obstacle_mid
        - collision_penalty
        - crash_penalty
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    return reward
