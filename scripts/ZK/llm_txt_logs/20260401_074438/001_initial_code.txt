def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    tilt_cos = kwargs["tilt_cos"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    dist_progress = prev_dist - curr_dist
    progress_reward = torch.tanh(dist_progress * 2.0)

    reach_reward = torch.exp(-1.5 * curr_dist)
    close_bonus = (curr_dist < 1.0).to(curr_dist.dtype) * torch.exp(-0.5 * curr_dist) * 8.0

    speed_penalty = torch.relu(v_norm - 4.0) ** 2
    smooth_penalty = action_diff + 0.15 * torch.norm(omega, dim=-1)

    altitude_penalty = torch.relu(torch.abs(z_pos - 2.0) - 0.35) ** 2

    obs_safe = torch.clamp(min_lidar_dist, min=0.0)
    obstacle_penalty = torch.exp(-6.0 * obs_safe)
    severe_obstacle_penalty = torch.relu(0.6 - min_lidar_dist) ** 2 * 20.0 + torch.relu(0.25 - min_lidar_dist) ** 2 * 120.0

    upright_reward = torch.clamp(tilt_cos, -1.0, 1.0)

    vel_norm_safe = v_norm.clamp_min(1e-6)
    move_dir = v / vel_norm_safe.unsqueeze(-1)
    heading_alignment = (x_body * move_dir).sum(dim=-1)
    heading_alignment = torch.clamp(heading_alignment, -1.0, 1.0)
    alignment_reward = torch.relu(v_norm - 0.3) * heading_alignment

    reward = (
        6.0 * progress_reward
        + 10.0 * reach_reward
        + close_bonus
        + 1.5 * upright_reward
        + 1.0 * alignment_reward
        - 0.8 * speed_penalty
        - 0.6 * smooth_penalty
        - 2.5 * altitude_penalty
        - 3.0 * obstacle_penalty
        - severe_obstacle_penalty
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    return reward
