def compute_reward(**kwargs):
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    progress = prev_dist - curr_dist
    progress_reward = torch.tanh(8.0 * progress)

    dist_reward = torch.exp(-0.7 * curr_dist)
    near_goal_reward = torch.exp(-2.0 * curr_dist)
    success_bonus = (curr_dist < 0.8).to(curr_dist.dtype) * 200.0
    very_close_bonus = (curr_dist < 0.4).to(curr_dist.dtype) * 400.0

    speed_pref = torch.tanh(v_norm / 3.0)
    speed_penalty = torch.relu(v_norm - 4.5) ** 2

    omega_norm = torch.norm(omega, dim=-1)
    smooth_penalty = torch.clamp(action_diff + 0.2 * omega_norm, max=100.0)

    altitude_error = torch.abs(z_pos - 2.0)
    altitude_penalty = torch.relu(altitude_error - 0.35) ** 2

    obs = min_lidar_dist.clamp_min(0.0)
    obs_safe = obs.clamp_min(1e-6)

    obstacle_hard = 1.0 / (obs_safe ** 2)
    obstacle_harder = 1.0 / (obs_safe ** 4)
    obstacle_soft = torch.exp(-2.5 * obs)
    obstacle_medium = torch.exp(-8.0 * obs)
    obstacle_sharp = torch.exp(-20.0 * obs)

    collision_zone = (min_lidar_dist < 0.8).to(min_lidar_dist.dtype)
    near_collision_zone = (min_lidar_dist < 0.5).to(min_lidar_dist.dtype)
    severe_collision_zone = (min_lidar_dist < 0.25).to(min_lidar_dist.dtype)
    extreme_collision_zone = (min_lidar_dist < 0.12).to(min_lidar_dist.dtype)

    obstacle_penalty = (
        3.0 * obstacle_soft
        + 10.0 * obstacle_medium
        + 25.0 * obstacle_sharp
        + 0.2 * obstacle_hard
        + 0.02 * obstacle_harder
        + collision_zone * (40.0 + 20.0 * obstacle_medium)
        + near_collision_zone * (150.0 + 80.0 * obstacle_sharp + 10.0 * obstacle_hard)
        + severe_collision_zone * (500.0 + 200.0 * obstacle_medium + 40.0 * obstacle_hard)
        + extreme_collision_zone * (1500.0 + 600.0 * obstacle_sharp + 120.0 * obstacle_hard)
    )

    vel_safe = v_norm.clamp_min(1e-6)
    move_dir = v / vel_safe.unsqueeze(-1)
    heading_alignment = (x_body * move_dir).sum(dim=-1)
    heading_alignment = torch.clamp(heading_alignment, -1.0, 1.0)
    alignment_reward = torch.relu(v_norm - 0.15) * heading_alignment

    goal_region = (curr_dist < 1.5).to(curr_dist.dtype) * 30.0
    terminal_goal = (curr_dist < 0.6).to(curr_dist.dtype) * 500.0

    reward = (
        25.0 * progress_reward
        + 12.0 * dist_reward
        + 10.0 * near_goal_reward
        + success_bonus
        + very_close_bonus
        + goal_region
        + terminal_goal
        + 1.0 * alignment_reward
        + 0.4 * speed_pref
        - 0.25 * speed_penalty
        - 0.15 * smooth_penalty
        - 1.5 * altitude_penalty
        - obstacle_penalty
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    return reward
