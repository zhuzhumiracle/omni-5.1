def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    tilt_cos = kwargs["tilt_cos"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    eps = 1e-6
    target_altitude = 2.0

    target_vec = target_pos - curr_pos
    target_dir = target_vec / torch.norm(target_vec, dim=-1, keepdim=True).clamp_min(eps)
    forward_align = (x_body * target_dir).sum(dim=-1).clamp(-1.0, 1.0)

    progress_reward = torch.tanh(2.5 * (prev_dist - curr_dist))
    proximity_reward = torch.exp(-0.28 * curr_dist)
    forward_reward = 0.55 * torch.clamp(forward_align, 0.0, 1.0) * torch.exp(-0.12 * curr_dist)
    goal_bonus = (curr_dist < 0.75).to(curr_dist.dtype) * 2000.0
    goal_shaping = (curr_dist < 6.0).to(curr_dist.dtype) * 8.0 * torch.exp(-0.55 * curr_dist)

    omega_norm = torch.norm(omega, dim=-1)
    smooth_penalty = 0.06 * action_diff + 0.05 * omega_norm + 0.01 * torch.square(omega_norm)

    global_tilt_penalty = 0.65 * torch.relu(0.92 - tilt_cos) + 0.25 * torch.square(torch.relu(0.96 - tilt_cos))

    altitude_reg = -0.12 * torch.square(z_pos - target_altitude)

    safe_dist = min_lidar_dist.clamp_min(eps)
    near_mask = (min_lidar_dist < 2.0).to(curr_dist.dtype)
    critical_mask = (min_lidar_dist < 0.4).to(curr_dist.dtype)

    obstacle_penalty = (
        0.55 * torch.exp(-1.4 * safe_dist)
        + 1.25 * torch.exp(-4.5 * safe_dist)
        + 2.5 * torch.exp(-10.0 * safe_dist)
        + critical_mask * 8.0
    )

    v_z = v[:, 2]
    near_speed_penalty = near_mask * (0.18 * v_norm + 0.08 * torch.square(v_norm))
    near_vz_penalty = near_mask * (0.12 * torch.abs(v_z) + 0.05 * torch.square(v_z))
    near_omega_penalty = near_mask * (0.04 * omega_norm + 0.02 * torch.square(omega_norm))

    speed_penalty = 0.05 * torch.relu(v_norm - 4.0) + 0.03 * torch.square(torch.relu(v_norm - 4.0))

    is_flipped = tilt_cos < 0.18
    is_out_of_bounds = (z_pos < 0.8) | (z_pos > 3.5) | (v_norm > 8.0)
    death_penalty = (is_flipped | is_out_of_bounds).to(curr_dist.dtype) * (25.0 + 10.0 * critical_mask)

    reward = (
        4.5 * progress_reward
        + 1.8 * proximity_reward
        + 0.8 * goal_shaping
        + forward_reward
        + goal_bonus
        + altitude_reg
        - smooth_penalty
        - global_tilt_penalty
        - obstacle_penalty
        - near_speed_penalty
        - near_vz_penalty
        - near_omega_penalty
        - speed_penalty
        - death_penalty
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    return reward
