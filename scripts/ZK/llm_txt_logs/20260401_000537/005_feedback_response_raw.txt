def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    dist_progress = prev_dist - curr_dist
    progress_reward = torch.tanh(6.0 * dist_progress)
    distance_reward = torch.exp(-0.35 * curr_dist)
    approach_reward = 14.0 * progress_reward + 10.0 * distance_reward

    success_1 = (curr_dist < 2.5).float()
    success_2 = (curr_dist < 1.5).float()
    success_3 = (curr_dist < 0.75).float()
    success_4 = (curr_dist < 0.35).float()
    success_bonus = 40.0 * success_1 + 120.0 * success_2 + 300.0 * success_3 + 800.0 * success_4

    dir_vec = target_pos - curr_pos
    dir_norm = torch.norm(dir_vec, dim=-1, keepdim=True).clamp_min(1e-6)
    dir_unit = dir_vec / dir_norm

    v_unit = v / v_norm.unsqueeze(-1).clamp_min(1e-6)
    align_target = torch.sum(x_body * dir_unit, dim=-1).clamp(-1.0, 1.0)
    vel_align_target = torch.sum(v_unit * dir_unit, dim=-1).clamp(-1.0, 1.0)
    heading_reward = 1.5 * align_target + 1.0 * vel_align_target

    desired_altitude = 2.0
    altitude_error = torch.abs(z_pos - desired_altitude)
    altitude_reward = torch.exp(-1.8 * altitude_error)
    altitude_penalty = torch.relu(altitude_error - 0.35)

    speed_penalty = torch.relu(v_norm - 4.5)
    omega_norm = torch.norm(omega, dim=-1)
    smooth_penalty = 0.04 * action_diff + 0.03 * omega_norm

    lidar = min_lidar_dist.clamp_min(1e-6)
    obstacle_reward = -(
        0.8 * torch.exp(-1.0 * lidar)
        + 2.0 * torch.exp(-2.5 * lidar)
        + 8.0 * torch.exp(-6.0 * lidar)
        + 30.0 * torch.exp(-12.0 * lidar)
        + 120.0 * torch.exp(-25.0 * lidar)
        + 500.0 * torch.exp(-50.0 * lidar)
        + 2000.0 * torch.exp(-100.0 * lidar)
    )

    collision_penalty = (
        -20.0 * torch.relu(0.6 - min_lidar_dist)
        -80.0 * torch.relu(0.35 - min_lidar_dist)
        -300.0 * torch.relu(0.20 - min_lidar_dist)
        -1200.0 * torch.relu(0.10 - min_lidar_dist)
        -5000.0 * torch.relu(0.05 - min_lidar_dist)
        -20000.0 * torch.relu(0.02 - min_lidar_dist)
        -80000.0 * torch.relu(0.01 - min_lidar_dist)
        -200000.0 * torch.exp(-120.0 * lidar)
    )

    reward = (
        approach_reward
        + success_bonus
        + 1.2 * heading_reward
        + 0.8 * altitude_reward
        - 2.0 * altitude_penalty
        - 0.6 * speed_penalty
        - smooth_penalty
        + obstacle_reward
        + collision_penalty
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    return reward
