def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    dist_progress = prev_dist - curr_dist
    progress_reward = torch.tanh(10.0 * dist_progress)

    target_proximity = torch.exp(-0.7 * curr_dist)
    target_reward = 18.0 * target_proximity

    success_close = (curr_dist < 1.0).float()
    success_near = (curr_dist < 0.5).float()
    success_bonus = 60.0 * success_close + 140.0 * success_near

    dir_vec = target_pos - curr_pos
    dir_norm = torch.norm(dir_vec, dim=-1, keepdim=True).clamp_min(1e-6)
    dir_unit = dir_vec / dir_norm

    v_dir = v / v_norm.unsqueeze(-1).clamp_min(1e-6)
    align_to_target = torch.sum(x_body * dir_unit, dim=-1)
    motion_to_target = torch.sum(v_dir * dir_unit, dim=-1)
    direction_reward = 0.5 * torch.clamp(align_to_target, -1.0, 1.0) + 0.5 * torch.clamp(motion_to_target, -1.0, 1.0)

    omega_norm = torch.norm(omega, dim=-1)
    smooth_penalty = 0.05 * action_diff + 0.02 * omega_norm

    desired_altitude = 2.0
    altitude_error = torch.abs(z_pos - desired_altitude)
    altitude_reward = torch.exp(-1.8 * altitude_error)
    altitude_penalty = torch.relu(altitude_error - 0.5)

    speed_penalty = torch.square(torch.relu(v_norm - 4.5))

    lidar = min_lidar_dist.clamp_min(1e-6)
    obstacle_soft_penalty = 3.0 * torch.exp(-4.0 * lidar) + 8.0 * torch.exp(-12.0 * lidar)
    obstacle_hard_penalty = 40.0 * torch.relu(0.6 - min_lidar_dist) ** 2 + 180.0 * torch.relu(0.3 - min_lidar_dist) ** 2 + 900.0 * torch.relu(0.12 - min_lidar_dist) ** 2 + 2500.0 * torch.exp(-35.0 * lidar)

    reward = (
        14.0 * progress_reward
        + target_reward
        + success_bonus
        + 1.2 * direction_reward
        + 0.8 * altitude_reward
        - 1.8 * altitude_penalty
        - 0.25 * speed_penalty
        - smooth_penalty
        - obstacle_soft_penalty
        - obstacle_hard_penalty
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    return reward
