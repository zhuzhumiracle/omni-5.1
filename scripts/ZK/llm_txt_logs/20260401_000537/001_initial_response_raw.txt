def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    tilt_cos = kwargs["tilt_cos"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    dist_progress = prev_dist - curr_dist
    progress_reward = torch.tanh(2.0 * dist_progress)

    target_reach = torch.exp(-1.5 * curr_dist) * 8.0

    altitude_error = torch.abs(z_pos - 2.0)
    altitude_reward = -torch.clamp(altitude_error - 0.25, min=0.0) - 0.25 * torch.square(altitude_error)

    speed_penalty = torch.square(torch.relu(v_norm - 4.0))

    smooth_penalty = 0.05 * action_diff + 0.02 * torch.norm(omega, dim=-1)

    upright_reward = torch.clamp(tilt_cos, min=0.0, max=1.0)

    dir_vec = target_pos - curr_pos
    dir_norm = torch.norm(dir_vec, dim=-1, keepdim=True).clamp_min(1e-6)
    dir_unit = dir_vec / dir_norm
    vel_norm = v_norm.clamp_min(1e-6).unsqueeze(-1)
    vel_dir = v / vel_norm
    align_reward = torch.sum(x_body * dir_unit, dim=-1)
    motion_align = torch.sum(vel_dir * dir_unit, dim=-1)
    align_reward = 0.5 * align_reward + 0.5 * motion_align

    obstacle_penalty = torch.exp(-6.0 * min_lidar_dist)
    severe_obstacle_penalty = torch.square(torch.relu(0.6 - min_lidar_dist)) * 20.0 + 2.0 * obstacle_penalty

    close_goal_bonus = (curr_dist < 1.0).float() * 3.0 + (curr_dist < 0.5).float() * 8.0

    reward = (
        4.0 * progress_reward
        + target_reach
        + 1.2 * upright_reward
        + 0.8 * align_reward
        + 0.5 * altitude_reward
        - 0.4 * speed_penalty
        - smooth_penalty
        - severe_obstacle_penalty
        - 0.5 * obstacle_penalty
        + close_goal_bonus
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    return reward
