def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    tilt_cos = kwargs["tilt_cos"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    eps = 1e-6

    progress = prev_dist - curr_dist
    progress_reward = torch.tanh(2.0 * progress)

    reach_reward = torch.exp(-4.0 * curr_dist)

    target_dir = target_pos - curr_pos
    target_dir_norm = torch.norm(target_dir, dim=-1).clamp_min(eps)
    target_dir_unit = target_dir / target_dir_norm.unsqueeze(-1)
    forward_align = (x_body * target_dir_unit).sum(dim=-1).clamp(-1.0, 1.0)
    align_reward = 0.5 * (forward_align + 1.0)

    height_error = torch.abs(z_pos - 2.0)
    height_reward = torch.exp(-1.5 * height_error)

    speed_penalty = torch.tanh(0.5 * v_norm)

    omega_norm = torch.norm(omega, dim=-1)
    omega_penalty = torch.tanh(0.25 * omega_norm)

    action_penalty = torch.tanh(action_diff)

    upright_reward = 0.5 * (tilt_cos.clamp(-1.0, 1.0) + 1.0)

    obstacle_penalty = 1.0 / (min_lidar_dist.clamp_min(eps) ** 2)

    close_target_bonus = torch.exp(-10.0 * curr_dist)

    reward = (
        2.5 * progress_reward
        + 3.0 * reach_reward
        + 0.6 * align_reward
        + 0.8 * height_reward
        + 0.8 * upright_reward
        + 2.0 * close_target_bonus
        - 1.2 * speed_penalty
        - 0.4 * omega_penalty
        - 0.5 * action_penalty
        - 0.8 * height_error.clamp_max(3.0) / 3.0
        - 1.5 * torch.tanh(obstacle_penalty)
    )

    reward = torch.nan_to_num(reward, nan=0.0, posinf=0.0, neginf=0.0)
    return reward
