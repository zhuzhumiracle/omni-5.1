def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    tilt_cos = kwargs["tilt_cos"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    dist_progress = prev_dist - curr_dist
    progress_term = torch.tanh(2.5 * dist_progress)

    reach_term = torch.exp(-1.5 * curr_dist)
    close_bonus = (curr_dist < 1.0).to(curr_dist.dtype) * 5.0
    very_close_bonus = (curr_dist < 0.35).to(curr_dist.dtype) * 25.0

    z_error = z_pos - 2.0
    z_penalty = torch.square(z_error) + 0.5 * torch.relu(torch.abs(z_error) - 0.35)

    speed_penalty = torch.square(torch.relu(v_norm - 4.0))
    smooth_penalty = 0.15 * action_diff + 0.05 * torch.sum(torch.square(omega), dim=-1)

    upright_bonus = torch.clamp(tilt_cos, min=-1.0, max=1.0)
    upright_penalty = torch.relu(0.7 - upright_bonus)

    v_norm_safe = v_norm.clamp_min(1e-6)
    v_dir = v / v_norm_safe.unsqueeze(-1)
    align_term = torch.sum(x_body * v_dir, dim=-1)
    align_term = torch.where(v_norm > 0.3, align_term, torch.zeros_like(align_term))

    obstacle_soft = torch.exp(-2.0 * min_lidar_dist)
    obstacle_hard = torch.relu(0.5 - min_lidar_dist)
    obstacle_critical = torch.relu(0.2 - min_lidar_dist)
    obstacle_penalty = 4.0 * obstacle_soft + 10.0 * torch.square(obstacle_hard) + 80.0 * torch.square(obstacle_critical)

    reward = (
        3.0 * progress_term
        + 1.5 * reach_term
        + close_bonus
        + very_close_bonus
        - 1.2 * z_penalty
        - 0.8 * speed_penalty
        - smooth_penalty
        + 0.8 * upright_bonus
        + 0.3 * align_term
        - obstacle_penalty
        - 0.6 * upright_penalty
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    return reward
