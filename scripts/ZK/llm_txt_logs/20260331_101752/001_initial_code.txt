def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    tilt_cos = kwargs["tilt_cos"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    dist_progress = prev_dist - curr_dist
    progress_reward = 4.0 * torch.tanh(2.0 * dist_progress)

    goal_reward = 8.0 * torch.exp(-1.5 * curr_dist) + 10.0 * (curr_dist < 0.5).float()

    altitude_error = torch.abs(z_pos - 2.0)
    altitude_reward = -1.5 * torch.square(torch.tanh(altitude_error))

    obstacle_safe = torch.clamp(min_lidar_dist, min=0.0)
    obstacle_penalty = -6.0 * torch.exp(-4.0 * obstacle_safe) - 20.0 * torch.relu(0.6 - obstacle_safe).pow(2) / (obstacle_safe + 1e-3)

    speed_penalty = -0.4 * torch.square(torch.relu(v_norm - 4.0)) - 0.05 * torch.square(v_norm)
    smooth_penalty = -0.15 * action_diff - 0.03 * torch.norm(omega, dim=-1)

    upright_reward = 1.2 * tilt_cos

    target_dir = target_pos - curr_pos
    target_dir_norm = torch.norm(target_dir, dim=-1, keepdim=True).clamp_min(1e-6)
    target_dir_unit = target_dir / target_dir_norm
    v_norm_safe = v_norm.clamp_min(1e-6).unsqueeze(-1)
    v_dir = v / v_norm_safe
    align_reward = 0.8 * torch.sum(x_body * target_dir_unit, dim=-1) + 0.4 * torch.sum(v_dir * target_dir_unit, dim=-1)

    reward = (
        progress_reward
        + goal_reward
        + altitude_reward
        + obstacle_penalty
        + speed_penalty
        + smooth_penalty
        + upright_reward
        + align_reward
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    return reward
