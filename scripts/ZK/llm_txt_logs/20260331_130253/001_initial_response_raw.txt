def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    tilt_cos = kwargs["tilt_cos"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    dist_progress = prev_dist - curr_dist
    progress_reward = torch.tanh(2.0 * dist_progress)

    goal_reach_bonus = torch.exp(-1.5 * curr_dist) * 8.0
    close_goal_bonus = (curr_dist < 0.5).to(curr_dist.dtype) * 20.0

    height_error = torch.abs(z_pos - 2.0)
    height_reward = -torch.square(torch.clamp(height_error, max=3.0))

    obstacle_soft = -torch.exp(-3.0 * min_lidar_dist)
    obstacle_hard = -4.0 / torch.clamp(min_lidar_dist * min_lidar_dist + 1e-3, min=1e-3)
    obstacle_penalty = obstacle_soft + 0.1 * obstacle_hard

    speed_penalty = -torch.square(torch.relu(v_norm - 3.5))
    smooth_penalty = -0.25 * torch.tanh(action_diff) - 0.05 * torch.norm(omega, dim=-1)

    upright_reward = 1.5 * tilt_cos

    target_vec = target_pos - curr_pos
    target_dir = target_vec / torch.norm(target_vec, dim=-1, keepdim=True).clamp_min(1e-6)
    heading_align = (x_body * target_dir).sum(dim=-1)
    heading_reward = 0.5 * heading_align

    reward = (
        6.0 * progress_reward
        + goal_reach_bonus
        + close_goal_bonus
        + 1.5 * height_reward
        + 3.0 * obstacle_penalty
        + 0.8 * speed_penalty
        + smooth_penalty
        + upright_reward
        + heading_reward
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    return reward
