def compute_reward(**kwargs):
    curr_pos = kwargs.get("curr_pos", None)
    target_pos = kwargs.get("target_pos", None)
    curr_dist = kwargs.get("curr_dist", None)
    prev_dist = kwargs.get("prev_dist", None)
    v = kwargs.get("v", None)
    v_norm = kwargs.get("v_norm", None)
    omega = kwargs.get("omega", None)
    min_lidar_dist = kwargs.get("min_lidar_dist", None)
    action_diff = kwargs.get("action_diff", None)
    z_pos = kwargs.get("z_pos", None)
    tilt_cos = kwargs.get("tilt_cos", None)
    x_body = kwargs.get("x_body", None)

    eps = 1e-6

    target_vec = target_pos - curr_pos
    target_dist = torch.norm(target_vec, dim=-1).clamp_min(eps)
    target_dir = target_vec / target_dist.unsqueeze(-1)
    forward_align = (x_body * target_dir).sum(dim=-1).clamp(-1.0, 1.0)

    v_norm_safe = v_norm.clamp_min(0.0)
    omega_norm = torch.norm(omega, dim=-1)
    vz = v[:, 2]
    vz_abs = torch.abs(vz)

    progress = prev_dist - curr_dist
    progress_reward = 4.0 * torch.tanh(2.5 * progress)
    proximity_reward = 2.2 * torch.exp(-0.22 * curr_dist)
    goal_bonus = (curr_dist < 6.0).float() * 10.0
    forward_reward = 0.9 * torch.relu(forward_align) * torch.tanh(v_norm_safe / 2.0)

    speed_target = 2.2
    speed_band_penalty = -0.30 * torch.square((v_norm_safe - speed_target) / speed_target)
    low_speed_penalty = -0.70 * torch.relu(0.85 - v_norm_safe)
    high_speed_penalty = -0.22 * torch.relu(v_norm_safe - 3.5)

    safe_dist = min_lidar_dist.clamp_min(eps)
    near_mask = (min_lidar_dist < 2.0).float()
    close_mask = (min_lidar_dist < 0.8).float()
    critical_mask = (min_lidar_dist < 0.4).float()

    obstacle_penalty = (
        0.65 * torch.exp(-1.2 * safe_dist)
        + 1.25 * torch.exp(-3.5 * safe_dist)
        + 0.60 * torch.reciprocal(safe_dist + 0.12)
    )

    near_motion_penalty = near_mask * (
        0.30 * v_norm_safe
        + 0.22 * vz_abs
        + 0.12 * omega_norm
        + 0.08 * torch.square(v_norm_safe)
    )
    near_upward_penalty = near_mask * 0.45 * torch.relu(vz)
    near_xy_penalty = close_mask * (0.10 * torch.norm(v[:, :2], dim=-1) + 0.06 * torch.square(torch.norm(v[:, :2], dim=-1)))

    critical_motion_penalty = critical_mask * (
        0.75 * v_norm_safe
        + 0.45 * vz_abs
        + 0.22 * omega_norm
        + 0.12 * torch.square(v_norm_safe)
    )
    collision_penalty = critical_mask * 8.0

    smooth_penalty = 0.10 * action_diff + 0.08 * omega_norm
    tilt_penalty = 0.70 * torch.relu(0.88 - tilt_cos) + 0.25 * torch.square(torch.relu(0.92 - tilt_cos))
    altitude_reg = 0.18 * torch.square(z_pos - 2.0)

    is_flipped = tilt_cos < 0.20
    is_out_of_bounds = (z_pos < 0.8) | (z_pos > 3.5) | (v_norm_safe > 8.0)
    death_penalty = (is_flipped | is_out_of_bounds).float() * 20.0

    close_goal_bonus = 2.0 * torch.exp(-0.10 * curr_dist)

    reward = (
        progress_reward
        + proximity_reward
        + goal_bonus
        + close_goal_bonus
        + forward_reward
        + speed_band_penalty
        + low_speed_penalty
        + high_speed_penalty
        - smooth_penalty
        - tilt_penalty
        - altitude_reg
        - obstacle_penalty
        - near_motion_penalty
        - near_upward_penalty
        - near_xy_penalty
        - critical_motion_penalty
        - collision_penalty
        - death_penalty
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = reward.view(-1)
    return reward
