def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    tilt_cos = kwargs["tilt_cos"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    eps = 1e-6

    target_vec = target_pos - curr_pos
    target_dir_norm = torch.norm(target_vec, dim=-1).clamp_min(eps)
    target_dir = target_vec / target_dir_norm.unsqueeze(-1)

    progress = prev_dist - curr_dist
    progress_reward = 12.0 * torch.tanh(6.0 * progress)

    proximity_reward = 18.0 * torch.exp(-2.5 * curr_dist)

    success_bonus = 40.0 * (curr_dist < 0.5).to(curr_dist.dtype)

    heading_align = (x_body * target_dir).sum(dim=-1).clamp(-1.0, 1.0)
    heading_reward = 3.0 * torch.clamp(heading_align, min=0.0)

    vel_toward_target = (v * target_dir).sum(dim=-1)
    motion_reward = 2.0 * torch.tanh(vel_toward_target)

    height_error = z_pos - 2.0
    height_reward = 1.5 * torch.exp(-1.25 * height_error * height_error)

    speed_penalty = -0.12 * torch.tanh(v_norm)

    omega_norm = torch.norm(omega, dim=-1)
    angular_penalty = -0.10 * torch.tanh(omega_norm)

    smooth_penalty = -0.15 * torch.tanh(action_diff)

    tilt_penalty = -2.0 * torch.clamp(1.0 - tilt_cos.clamp(-1.0, 1.0), min=0.0)

    safe_dist = min_lidar_dist.clamp_min(0.0)
    collision_zone = torch.clamp(0.8 - safe_dist, min=0.0)

    obstacle_penalty = -35.0 / (safe_dist.clamp_min(0.03))
    near_obstacle_penalty = -25.0 * torch.exp(-10.0 * safe_dist)
    critical_obstacle_penalty = -80.0 * (collision_zone / 0.8) * (1.0 / (safe_dist.clamp_min(0.01)))

    collision_penalty = -120.0 * (safe_dist < 0.08).to(curr_dist.dtype)

    reward = (
        progress_reward
        + proximity_reward
        + success_bonus
        + heading_reward
        + motion_reward
        + height_reward
        + speed_penalty
        + angular_penalty
        + smooth_penalty
        + tilt_penalty
        + obstacle_penalty
        + near_obstacle_penalty
        + critical_obstacle_penalty
        + collision_penalty
    )

    reward = reward.view(-1)
    return reward
