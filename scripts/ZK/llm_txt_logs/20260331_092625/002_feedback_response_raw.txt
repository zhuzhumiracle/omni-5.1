def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    tilt_cos = kwargs["tilt_cos"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    eps = 1e-6

    progress = prev_dist - curr_dist
    progress_reward = 3.0 * torch.tanh(5.0 * progress)

    proximity_reward = 4.0 * torch.exp(-2.0 * curr_dist)

    success_bonus = 8.0 * (curr_dist < 0.5).to(curr_dist.dtype)

    target_dir = target_pos - curr_pos
    target_dir_norm = torch.norm(target_dir, dim=-1).clamp_min(eps)
    target_dir_unit = target_dir / target_dir_norm.unsqueeze(-1)
    heading_align = (x_body * target_dir_unit).sum(dim=-1).clamp(-1.0, 1.0)
    heading_reward = 1.5 * 0.5 * (heading_align + 1.0)

    velocity_toward_target = (v * target_dir_unit).sum(dim=-1)
    motion_reward = 0.5 * torch.tanh(velocity_toward_target)

    height_error = z_pos - 2.0
    height_reward = torch.exp(-1.2 * height_error * height_error)

    speed_penalty = -0.15 * torch.tanh(v_norm)

    omega_norm = torch.norm(omega, dim=-1)
    angular_penalty = -0.08 * torch.tanh(omega_norm)

    smooth_penalty = -0.12 * torch.tanh(action_diff)

    tilt_penalty = -0.8 * torch.clamp(1.0 - tilt_cos.clamp(-1.0, 1.0), min=0.0)

    safe_dist = min_lidar_dist.clamp_min(0.0)
    collision_barrier = -12.0 / (safe_dist + 0.05)
    near_collision_penalty = -6.0 * torch.clamp(0.35 - min_lidar_dist, min=0.0) / 0.35

    reward = (
        progress_reward
        + proximity_reward
        + success_bonus
        + heading_reward
        + motion_reward
        + 0.8 * height_reward
        + speed_penalty
        + angular_penalty
        + smooth_penalty
        + tilt_penalty
        + collision_barrier
        + near_collision_penalty
    )

    reward = reward.view(-1)
    return reward
