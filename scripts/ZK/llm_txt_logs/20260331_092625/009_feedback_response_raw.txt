def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    tilt_cos = kwargs["tilt_cos"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    eps = 1e-6

    target_vec = target_pos - curr_pos
    target_dir = target_vec / torch.norm(target_vec, dim=-1).clamp_min(eps).unsqueeze(-1)

    progress = prev_dist - curr_dist
    progress_reward = 200.0 * torch.tanh(12.0 * progress)

    proximity_reward = 150.0 * torch.exp(-3.5 * curr_dist)
    near_target_bonus = 100.0 * torch.exp(-18.0 * curr_dist)
    success_bonus = 800.0 * (curr_dist < 0.35).to(curr_dist.dtype)

    heading_align = (x_body * target_dir).sum(dim=-1).clamp(-1.0, 1.0)
    heading_reward = 30.0 * torch.clamp(heading_align, min=0.0)

    vel_toward_target = (v * target_dir).sum(dim=-1)
    motion_reward = 18.0 * torch.tanh(vel_toward_target)

    height_error = z_pos - 2.0
    height_reward = 6.0 * torch.exp(-1.0 * height_error * height_error)

    speed_penalty = -0.05 * torch.tanh(v_norm)
    omega_norm = torch.norm(omega, dim=-1)
    angular_penalty = -0.15 * torch.tanh(omega_norm)
    smooth_penalty = -0.25 * torch.tanh(action_diff)

    tilt_penalty = -35.0 * torch.clamp(1.0 - tilt_cos.clamp(-1.0, 1.0), min=0.0)

    safe_dist = min_lidar_dist.clamp_min(eps)
    inv_dist = 1.0 / safe_dist
    inv_dist2 = inv_dist * inv_dist
    obstacle_soft_penalty = -15.0 * torch.exp(-4.0 * safe_dist)
    obstacle_hard_penalty = -120.0 * inv_dist2
    obstacle_critical_penalty = -300.0 * inv_dist2 * inv_dist
    collision_penalty = -5000.0 * (safe_dist < 0.20).to(curr_dist.dtype)
    near_collision_penalty = -800.0 * torch.clamp(0.25 - safe_dist, min=0.0) * inv_dist2

    reward = (
        progress_reward
        + proximity_reward
        + near_target_bonus
        + success_bonus
        + heading_reward
        + motion_reward
        + height_reward
        + speed_penalty
        + angular_penalty
        + smooth_penalty
        + tilt_penalty
        + obstacle_soft_penalty
        + obstacle_hard_penalty
        + obstacle_critical_penalty
        + collision_penalty
        + near_collision_penalty
    )

    reward = reward.view(-1)
    return reward
