def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    tilt_cos = kwargs["tilt_cos"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    eps = 1e-6

    progress = prev_dist - curr_dist
    progress_reward = torch.tanh(2.0 * progress)

    reach_reward = torch.exp(-4.0 * curr_dist)

    target_dir = target_pos - curr_pos
    target_dir_norm = torch.norm(target_dir, dim=-1).clamp_min(eps)
    target_dir_unit = target_dir / target_dir_norm.unsqueeze(-1)
    align = (x_body * target_dir_unit).sum(dim=-1).clamp(-1.0, 1.0)
    align_reward = 0.5 * (align + 1.0)

    height_error = z_pos - 2.0
    height_reward = torch.exp(-1.5 * height_error * height_error)

    speed_penalty = -0.15 * torch.tanh(v_norm)

    obs_safe = min_lidar_dist.clamp_min(0.0)
    obstacle_penalty = -torch.exp(-8.0 * obs_safe) * 3.0 - torch.clamp(0.35 - min_lidar_dist, min=0.0) * 10.0

    smooth_penalty = -0.1 * torch.tanh(action_diff)

    tilt_reward = 0.4 * tilt_cos.clamp(-1.0, 1.0)

    omega_norm = torch.norm(omega, dim=-1)
    angular_penalty = -0.05 * torch.tanh(omega_norm)

    velocity_along_target = (v * target_dir_unit).sum(dim=-1)
    directional_motion_reward = 0.1 * torch.tanh(velocity_along_target)

    close_bonus = torch.exp(-10.0 * curr_dist) * 2.0

    reward = (
        1.2 * progress_reward
        + 1.5 * reach_reward
        + 0.3 * align_reward
        + 0.8 * height_reward
        + speed_penalty
        + obstacle_penalty
        + smooth_penalty
        + tilt_reward
        + angular_penalty
        + directional_motion_reward
        + close_bonus
    )

    reward = reward.view(-1)
    return reward
