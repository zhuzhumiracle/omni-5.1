def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    tilt_cos = kwargs["tilt_cos"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    eps = 1e-6

    progress = prev_dist - curr_dist
    progress_reward = torch.tanh(3.0 * progress)

    goal_reach = torch.exp(-2.5 * curr_dist)
    close_bonus = torch.exp(-0.7 * curr_dist) * (curr_dist < 1.5).float()

    target_dir = target_pos - curr_pos
    target_dir_norm = torch.norm(target_dir, dim=-1).clamp_min(eps)
    target_dir_unit = target_dir / target_dir_norm.unsqueeze(-1)
    align = (x_body * target_dir_unit).sum(dim=-1)
    align_reward = torch.clamp(align, -1.0, 1.0)

    height_error = torch.abs(z_pos - 2.0)
    height_reward = torch.exp(-1.5 * height_error)

    obstacle_soft = -torch.exp(-2.0 * min_lidar_dist)
    obstacle_hard = -torch.exp(-8.0 * min_lidar_dist.clamp_min(0.0))
    obstacle_very_close = -(0.5 / (min_lidar_dist.clamp_min(eps) ** 2))

    speed_penalty = torch.relu(v_norm - 4.0) ** 2
    smooth_penalty = action_diff + 0.2 * torch.norm(omega, dim=-1)

    upright_reward = torch.clamp(tilt_cos, -1.0, 1.0)

    reward = (
        3.0 * progress_reward
        + 2.5 * goal_reach
        + 1.5 * close_bonus
        + 0.8 * align_reward
        + 1.2 * height_reward
        + 1.0 * upright_reward
        + 1.5 * obstacle_soft
        + 3.0 * obstacle_hard
        + 0.5 * obstacle_very_close
        - 0.6 * speed_penalty
        - 0.4 * smooth_penalty
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    return reward
