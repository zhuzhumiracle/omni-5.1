def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    tilt_cos = kwargs["tilt_cos"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    progress = prev_dist - curr_dist
    progress_reward = torch.tanh(2.0 * progress)

    dist_reward = torch.exp(-0.35 * curr_dist)

    reach_reward = (curr_dist < 0.6).to(curr_dist.dtype) * (8.0 + 4.0 * torch.exp(-3.0 * curr_dist))
    close_bonus = (curr_dist < 2.0).to(curr_dist.dtype) * torch.exp(-1.5 * curr_dist)

    z_error = torch.abs(z_pos - 2.0)
    z_reward = -torch.square(z_error)

    speed_penalty = torch.square(torch.relu(v_norm - 4.0))

    omega_norm = torch.norm(omega, dim=-1)
    smooth_penalty = action_diff + 0.1 * omega_norm

    obstacle_penalty = -6.0 * torch.exp(-3.0 * torch.clamp(min_lidar_dist, min=0.0))
    severe_obstacle_penalty = -3.0 / torch.clamp(min_lidar_dist + 0.05, min=1e-6)

    forward_dir = v / torch.clamp(v_norm.unsqueeze(-1), min=1e-6)
    align_move = (x_body * forward_dir).sum(dim=-1)
    align_reward = torch.clamp(align_move, min=-1.0, max=1.0) * torch.tanh(v_norm)

    attitude_reward = tilt_cos

    reward = (
        4.0 * progress_reward
        + 1.5 * dist_reward
        + 2.5 * close_bonus
        + reach_reward
        + 0.8 * z_reward
        - 0.6 * speed_penalty
        - 0.2 * smooth_penalty
        + 0.7 * attitude_reward
        + 0.5 * align_reward
        + obstacle_penalty
        + severe_obstacle_penalty
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    return reward
