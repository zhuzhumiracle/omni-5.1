def compute_reward(**kwargs):
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    progress = prev_dist - curr_dist
    progress_reward = torch.tanh(5.0 * progress)

    proximity_reward = torch.exp(-0.8 * curr_dist)
    target_bonus = (curr_dist < 0.75).to(curr_dist.dtype) * (20.0 + 10.0 * torch.exp(-4.0 * curr_dist))
    near_goal_bonus = (curr_dist < 2.5).to(curr_dist.dtype) * 4.0 * torch.exp(-1.2 * curr_dist)

    z_error = torch.abs(z_pos - 2.0)
    altitude_reward = torch.exp(-1.5 * z_error)
    altitude_penalty = -torch.relu(z_error - 0.5)

    omega_norm = torch.norm(omega, dim=-1)
    smooth_penalty = torch.tanh(action_diff) + 0.25 * torch.tanh(omega_norm)

    speed_excess = torch.relu(v_norm - 5.0)
    speed_penalty = torch.square(speed_excess)

    v_dir = v / v_norm.unsqueeze(-1).clamp_min(1e-6)
    alignment = (x_body * v_dir).sum(dim=-1)
    alignment_reward = torch.tanh(2.0 * alignment) * torch.tanh(v_norm)

    safe_lidar = min_lidar_dist.clamp_min(1e-6)
    obstacle_barrier = -2.0 / safe_lidar
    obstacle_near_penalty = -8.0 * torch.exp(-6.0 * torch.clamp(min_lidar_dist, min=0.0))
    collision_penalty = -(min_lidar_dist < 0.35).to(curr_dist.dtype) * (50.0 + 200.0 / safe_lidar)

    reward = (
        8.0 * progress_reward
        + 3.0 * proximity_reward
        + near_goal_bonus
        + target_bonus
        + 1.5 * altitude_reward
        + altitude_penalty
        + 1.0 * alignment_reward
        - 0.4 * smooth_penalty
        - 0.3 * speed_penalty
        + obstacle_barrier
        + obstacle_near_penalty
        + collision_penalty
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    return reward
