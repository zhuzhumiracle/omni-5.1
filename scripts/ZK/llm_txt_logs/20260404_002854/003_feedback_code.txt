def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    tilt_cos = kwargs["tilt_cos"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    dist_progress = prev_dist - curr_dist
    progress_reward = torch.tanh(2.5 * dist_progress)

    target_vec = target_pos - curr_pos
    target_dir = target_vec / torch.norm(target_vec, dim=-1).clamp_min(1e-6).unsqueeze(-1)
    forward_align = (x_body * target_dir).sum(dim=-1).clamp(-1.0, 1.0)

    vel_dir = v / v_norm.unsqueeze(-1).clamp_min(1e-6)
    vel_align = (x_body * vel_dir).sum(dim=-1).clamp(-1.0, 1.0)
    vel_align = torch.where(v_norm > 0.10, vel_align, torch.zeros_like(vel_align))

    proximity_reward = torch.exp(-0.25 * curr_dist)
    reach_bonus = (curr_dist < 0.75).to(curr_dist.dtype) * 10.0
    near_goal_bonus = torch.exp(-0.90 * curr_dist) * 2.5

    speed_term = 0.18 * torch.tanh(0.7 * v_norm) * torch.clamp(forward_align, 0.0, 1.0)
    align_term = 0.55 * torch.clamp(forward_align, -1.0, 1.0) + 0.15 * torch.clamp(vel_align, -1.0, 1.0)

    alt_err = z_pos - 2.0
    altitude_regularizer = -0.12 * torch.tanh(0.7 * alt_err * alt_err) - 0.08 * torch.relu(torch.abs(alt_err) - 1.8)

    omega_norm = torch.norm(omega, dim=-1)
    smooth_penalty = 0.035 * action_diff + 0.025 * omega_norm

    safe_dist = min_lidar_dist.clamp_min(1e-6)
    inv_dist = 1.0 / (safe_dist + 0.05)

    obstacle_core_penalty = 0.55 * torch.exp(-1.4 * safe_dist) + 1.25 * torch.exp(-5.5 * safe_dist) + 0.15 * inv_dist
    obstacle_hard_penalty = (min_lidar_dist < 0.35).to(min_lidar_dist.dtype) * (2.0 * torch.exp(-3.0 * safe_dist) + 1.5)
    collision_penalty = -5.0 * torch.exp(-12.0 * safe_dist)

    near_mask = (min_lidar_dist < 1.8).to(min_lidar_dist.dtype)
    very_near_mask = (min_lidar_dist < 0.9).to(min_lidar_dist.dtype)

    near_speed_penalty = near_mask * (0.18 * v_norm + 0.05 * torch.square(v_norm))
    near_vertical_penalty = near_mask * (0.12 * torch.abs(v[:, 2]) + 0.05 * torch.square(v[:, 2]))
    near_angular_penalty = near_mask * (0.08 * omega_norm + 0.03 * torch.square(omega_norm))
    near_tilt_penalty = near_mask * (0.16 * torch.relu(1.0 - tilt_cos))

    very_near_motion_penalty = very_near_mask * (
        0.10 * v_norm + 0.08 * torch.abs(v[:, 2]) + 0.06 * omega_norm + 0.08 * torch.relu(1.0 - tilt_cos)
    )

    reward = (
        3.5 * progress_reward
        + 1.2 * proximity_reward
        + near_goal_bonus
        + reach_bonus
        + align_term
        + speed_term
        + altitude_regularizer
        - smooth_penalty
        - obstacle_core_penalty
        - obstacle_hard_penalty
        - near_speed_penalty
        - near_vertical_penalty
        - near_angular_penalty
        - near_tilt_penalty
        - very_near_motion_penalty
        + collision_penalty
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    return reward
