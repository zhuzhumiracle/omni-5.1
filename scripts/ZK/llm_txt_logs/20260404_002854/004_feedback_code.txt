def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    tilt_cos = kwargs["tilt_cos"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    eps = 1e-6

    dist_progress = prev_dist - curr_dist
    progress_reward = torch.tanh(3.0 * dist_progress)

    goal_bonus = (curr_dist < 0.75).to(curr_dist.dtype) * 12.0
    proximity_reward = torch.exp(-0.30 * curr_dist) * 1.8
    shaping_goal = torch.exp(-0.90 * curr_dist) * 2.0

    target_vec = target_pos - curr_pos
    target_dir = target_vec / torch.norm(target_vec, dim=-1).clamp_min(eps).unsqueeze(-1)
    forward_align = (x_body * target_dir).sum(dim=-1).clamp(-1.0, 1.0)

    vel_dir = v / v_norm.unsqueeze(-1).clamp_min(eps)
    vel_align = (x_body * vel_dir).sum(dim=-1).clamp(-1.0, 1.0)
    vel_align = torch.where(v_norm > 0.10, vel_align, torch.zeros_like(vel_align))

    forward_reward = 0.65 * torch.clamp(forward_align, 0.0, 1.0)
    motion_reward = 0.18 * torch.tanh(0.6 * v_norm) * torch.clamp(forward_align, 0.0, 1.0)
    vel_track_reward = 0.12 * torch.clamp(vel_align, -1.0, 1.0)

    alt_err = z_pos - 2.0
    altitude_reg = -0.10 * torch.tanh(0.7 * torch.abs(alt_err)) - 0.06 * torch.relu(torch.abs(alt_err) - 1.4)

    omega_norm = torch.norm(omega, dim=-1)
    smooth_penalty = 0.030 * action_diff + 0.020 * omega_norm

    safe_dist = min_lidar_dist.clamp_min(eps)
    near_mask = (min_lidar_dist < 2.0).to(min_lidar_dist.dtype)
    very_near_mask = (min_lidar_dist < 1.0).to(min_lidar_dist.dtype)
    critical_mask = (min_lidar_dist < 0.35).to(min_lidar_dist.dtype)

    obstacle_repulsion = (
        0.40 * torch.exp(-0.9 * safe_dist)
        + 0.95 * torch.exp(-3.8 * safe_dist)
        + 1.30 * torch.exp(-10.0 * safe_dist)
        + 0.18 / (safe_dist + 0.08)
    )

    crash_penalty = critical_mask * (1.8 + 3.5 * torch.exp(-18.0 * safe_dist))
    obstacle_penalty = obstacle_repulsion + crash_penalty

    near_speed_penalty = near_mask * (0.16 * v_norm + 0.05 * torch.square(v_norm))
    near_vz_penalty = near_mask * (0.10 * torch.abs(v[:, 2]) + 0.04 * torch.square(v[:, 2]))
    near_omega_penalty = near_mask * (0.08 * omega_norm + 0.03 * torch.square(omega_norm))
    near_tilt_penalty = near_mask * (0.18 * torch.relu(1.0 - tilt_cos))

    very_near_motion_penalty = very_near_mask * (
        0.10 * v_norm + 0.08 * torch.abs(v[:, 2]) + 0.06 * omega_norm + 0.06 * torch.relu(1.0 - tilt_cos)
    )

    reward = (
        3.8 * progress_reward
        + 1.6 * proximity_reward
        + 1.2 * shaping_goal
        + goal_bonus
        + forward_reward
        + motion_reward
        + vel_track_reward
        + altitude_reg
        - smooth_penalty
        - obstacle_penalty
        - near_speed_penalty
        - near_vz_penalty
        - near_omega_penalty
        - near_tilt_penalty
        - very_near_motion_penalty
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    return reward
