def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    tilt_cos = kwargs["tilt_cos"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    dist_progress = prev_dist - curr_dist
    progress_reward = torch.tanh(3.0 * dist_progress)

    proximity_reward = torch.exp(-0.35 * curr_dist)
    reach_bonus = (curr_dist < 0.6).to(curr_dist.dtype) * 12.0
    close_goal_bonus = torch.exp(-1.2 * curr_dist) * 3.0

    target_dir = target_pos - curr_pos
    target_dir_norm = torch.norm(target_dir, dim=-1).clamp_min(1e-6)
    target_dir_unit = target_dir / target_dir_norm.unsqueeze(-1)
    forward_align = (x_body * target_dir_unit).sum(dim=-1).clamp(-1.0, 1.0)

    move_dir = v / v_norm.unsqueeze(-1).clamp_min(1e-6)
    velocity_align = (x_body * move_dir).sum(dim=-1)
    velocity_align = torch.where(v_norm > 0.15, velocity_align, torch.zeros_like(velocity_align))

    speed_reward = 0.25 * torch.tanh(0.8 * v_norm) * torch.clamp(forward_align, 0.0, 1.0)

    alt_err = z_pos - 2.0
    altitude_regularizer = -0.35 * torch.tanh(0.8 * alt_err * alt_err)
    altitude_soft_band = -0.15 * torch.relu(torch.abs(alt_err) - 1.2)

    omega_norm = torch.norm(omega, dim=-1)
    smooth_penalty = 0.06 * action_diff + 0.04 * omega_norm

    safe_dist = min_lidar_dist.clamp_min(1e-3)
    near_mask = (min_lidar_dist < 2.0).to(min_lidar_dist.dtype)
    very_near_mask = (min_lidar_dist < 0.7).to(min_lidar_dist.dtype)
    collision_mask = (min_lidar_dist < 0.25).to(min_lidar_dist.dtype)

    obstacle_penalty = 0.8 * torch.exp(-1.8 * safe_dist) + 2.5 * torch.exp(-5.0 * safe_dist)
    obstacle_speed_penalty = near_mask * (0.25 * v_norm + 0.08 * torch.square(v_norm))
    obstacle_vertical_penalty = near_mask * (0.20 * torch.abs(v[:, 2]) + 0.06 * torch.square(v[:, 2]))
    obstacle_rot_penalty = near_mask * (0.10 * omega_norm + 0.03 * torch.square(omega_norm))
    obstacle_tilt_penalty = near_mask * (0.18 * torch.relu(1.0 - tilt_cos))
    severe_collision_penalty = collision_mask * (-15.0)
    very_near_motion_penalty = very_near_mask * (
        0.18 * v_norm + 0.10 * torch.abs(v[:, 2]) + 0.08 * omega_norm + 0.10 * torch.relu(1.0 - tilt_cos)
    )

    stability_reward = 0.25 * torch.clamp(tilt_cos, -1.0, 1.0)

    reward = (
        4.0 * progress_reward
        + 1.8 * proximity_reward
        + 3.0 * close_goal_bonus
        + reach_bonus
        + 1.2 * torch.clamp(forward_align, -1.0, 1.0)
        + speed_reward
        + stability_reward
        + altitude_regularizer
        + altitude_soft_band
        - smooth_penalty
        - obstacle_penalty
        - obstacle_speed_penalty
        - obstacle_vertical_penalty
        - obstacle_rot_penalty
        - obstacle_tilt_penalty
        - very_near_motion_penalty
        + severe_collision_penalty
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    return reward
