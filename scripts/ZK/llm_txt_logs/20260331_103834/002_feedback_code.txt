def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    tilt_cos = kwargs.get("tilt_cos", torch.ones_like(curr_dist))
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    progress = prev_dist - curr_dist
    progress_reward = torch.tanh(4.0 * progress)

    close_target = torch.exp(-0.8 * curr_dist)
    near_goal_bonus = (curr_dist < 1.0).to(curr_dist.dtype) * 5.0
    success_bonus = (curr_dist < 0.5).to(curr_dist.dtype) * 50.0

    dist_penalty = -0.15 * torch.tanh(curr_dist / 10.0)

    z_error = torch.abs(z_pos - 2.0)
    z_penalty = -0.8 * torch.tanh(z_error)

    lidar_safe = min_lidar_dist.clamp_min(1e-6)
    obstacle_soft = -4.0 * torch.exp(-3.0 * lidar_safe)
    obstacle_hard = -30.0 * torch.relu(0.5 - min_lidar_dist).pow(2) - 80.0 * torch.relu(0.2 - min_lidar_dist).pow(2) - 200.0 * torch.relu(0.05 - min_lidar_dist).pow(2)

    speed_penalty = -0.10 * torch.relu(v_norm - 5.0).pow(2) - 0.03 * torch.tanh(v_norm / 8.0)

    omega_norm = torch.norm(omega, dim=-1)
    smooth_penalty = -0.03 * action_diff - 0.01 * torch.tanh(omega_norm)

    tilt_reward = 0.5 * tilt_cos

    target_vec = target_pos - curr_pos
    target_dist = target_vec.norm(dim=-1).clamp_min(1e-6)
    target_dir = target_vec / target_dist.unsqueeze(-1)
    body_align = (x_body * target_dir).sum(dim=-1)
    align_reward = 0.25 * body_align * torch.tanh(v_norm / 2.0)

    reward = (
        8.0 * progress_reward
        + 2.5 * close_target
        + near_goal_bonus
        + success_bonus
        + dist_penalty
        + z_penalty
        + obstacle_soft
        + obstacle_hard
        + speed_penalty
        + smooth_penalty
        + tilt_reward
        + align_reward
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    reward = reward.reshape(-1)
    return reward
