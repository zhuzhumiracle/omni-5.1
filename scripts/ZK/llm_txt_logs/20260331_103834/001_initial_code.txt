def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    tilt_cos = kwargs["tilt_cos"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    progress = prev_dist - curr_dist
    progress_reward = torch.tanh(2.0 * progress)

    goal_bonus = torch.exp(-2.5 * curr_dist)
    reach_bonus = (curr_dist < 0.6).to(curr_dist.dtype) * 10.0

    dist_term = -0.25 * torch.tanh(curr_dist / 8.0)

    z_error = torch.abs(z_pos - 2.0)
    z_penalty = -0.8 * torch.tanh(z_error)

    safe_lidar = min_lidar_dist.clamp_min(1e-6)
    obstacle_penalty = -3.0 * torch.exp(-5.0 * safe_lidar) - 8.0 * torch.relu(0.35 - min_lidar_dist).pow(2)

    speed_penalty = -0.15 * torch.relu(v_norm - 4.0).pow(2) - 0.05 * torch.tanh(v_norm / 6.0)

    omega_norm = torch.norm(omega, dim=-1)
    smooth_penalty = -0.05 * action_diff - 0.02 * torch.tanh(omega_norm)

    upright_reward = 0.8 * tilt_cos

    target_dir = target_pos - curr_pos
    target_dist = target_dir.norm(dim=-1).clamp_min(1e-6)
    target_dir_unit = target_dir / target_dist.unsqueeze(-1)
    heading_align = (x_body * target_dir_unit).sum(dim=-1)
    align_reward = 0.3 * heading_align * torch.tanh(v_norm)

    reward = (
        3.0 * progress_reward
        + 1.2 * goal_bonus
        + reach_bonus
        + dist_term
        + z_penalty
        + obstacle_penalty
        + speed_penalty
        + smooth_penalty
        + upright_reward
        + align_reward
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    reward = reward.reshape(-1)
    return reward
