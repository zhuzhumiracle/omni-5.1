def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    tilt_cos = kwargs["tilt_cos"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    eps = 1e-6

    progress = prev_dist - curr_dist
    progress_reward = 15.0 * torch.tanh(3.0 * progress)

    target_shaping = 3.0 * torch.exp(-0.25 * curr_dist)
    near_target_bonus = (curr_dist < 2.0).to(curr_dist.dtype) * (8.0 * torch.exp(-0.8 * curr_dist))
    success_bonus = (curr_dist < 0.5).to(curr_dist.dtype) * 300.0

    height_error = torch.abs(z_pos - 2.0)
    height_penalty = 4.0 * torch.square(torch.relu(height_error - 0.25))

    speed_penalty = 0.35 * torch.square(torch.relu(v_norm - 4.0)) + 0.03 * torch.square(v_norm)

    smooth_penalty = 0.12 * action_diff + 0.08 * torch.norm(omega, dim=-1)

    upright_reward = 1.2 * torch.clamp(tilt_cos, min=-1.0, max=1.0)

    move_dir = v / v_norm.unsqueeze(-1).clamp_min(eps)
    align_reward = 0.6 * torch.sum(x_body * move_dir, dim=-1) * torch.tanh(v_norm)

    obstacle_soft_penalty = 12.0 * torch.exp(-6.0 * torch.clamp(min_lidar_dist, min=0.0))
    obstacle_mid_penalty = 40.0 * torch.exp(-12.0 * torch.clamp(min_lidar_dist, min=0.0))
    obstacle_severe_penalty = 250.0 / (min_lidar_dist.clamp_min(eps) ** 2)
    collision_penalty = (min_lidar_dist < 0.4).to(min_lidar_dist.dtype) * (
        100.0 + 600.0 * torch.square(torch.clamp(0.4 - min_lidar_dist, min=0.0))
    )

    reward = (
        progress_reward
        + target_shaping
        + near_target_bonus
        + success_bonus
        - height_penalty
        - speed_penalty
        - smooth_penalty
        + upright_reward
        + align_reward
        - obstacle_soft_penalty
        - obstacle_mid_penalty
        - obstacle_severe_penalty
        - collision_penalty
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    return reward
