def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    tilt_cos = kwargs["tilt_cos"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    eps = 1e-6

    progress = prev_dist - curr_dist
    progress_reward = 8.0 * torch.tanh(2.0 * progress)

    distance_reward = 2.0 * torch.exp(-0.35 * curr_dist)

    close_bonus = (curr_dist < 1.0).to(curr_dist.dtype) * (10.0 * torch.exp(-2.0 * curr_dist))

    target_reach_bonus = (curr_dist < 0.5).to(curr_dist.dtype) * 200.0

    height_error = torch.abs(z_pos - 2.0)
    height_penalty = 3.0 * torch.square(torch.clamp(height_error - 0.15, min=0.0))

    speed_penalty = 0.6 * torch.square(torch.relu(v_norm - 4.5)) + 0.05 * torch.square(v_norm)

    smooth_penalty = 0.15 * action_diff + 0.05 * torch.norm(omega, dim=-1)

    upright_reward = 1.5 * tilt_cos

    obstacle_severity = torch.relu(0.9 - min_lidar_dist)
    obstacle_penalty = 25.0 * torch.exp(-8.0 * torch.clamp(min_lidar_dist, min=0.0)) + 120.0 * torch.square(obstacle_severity)

    danger_penalty = (min_lidar_dist < 0.35).to(min_lidar_dist.dtype) * (50.0 + 150.0 * torch.square(torch.clamp(0.35 - min_lidar_dist, min=0.0)))

    move_dir = v / v_norm.unsqueeze(-1).clamp_min(eps)
    align_reward = 0.5 * torch.sum(x_body * move_dir, dim=-1) * torch.tanh(v_norm)

    reward = (
        progress_reward
        + distance_reward
        + close_bonus
        + target_reach_bonus
        - height_penalty
        - speed_penalty
        - smooth_penalty
        + upright_reward
        - obstacle_penalty
        - danger_penalty
        + align_reward
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    return reward
