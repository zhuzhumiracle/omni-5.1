def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    tilt_cos = kwargs["tilt_cos"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    progress = prev_dist - curr_dist
    progress_reward = torch.tanh(2.0 * progress)

    dist_reward = torch.exp(-0.35 * curr_dist)
    goal_bonus = (curr_dist < 0.6).to(curr_dist.dtype) * 30.0
    near_goal_bonus = torch.exp(-1.5 * curr_dist) * 8.0

    z_error = z_pos - 2.0
    height_penalty = torch.square(z_error)

    speed_excess = torch.relu(v_norm - 4.0)
    speed_penalty = torch.square(speed_excess)

    smooth_penalty = torch.clamp(action_diff, max=10.0) + 0.1 * torch.clamp(torch.norm(omega, dim=-1), max=20.0)

    upright_reward = torch.clamp(tilt_cos, -1.0, 1.0)

    target_dir = target_pos - curr_pos
    target_dir_norm = torch.norm(target_dir, dim=-1).clamp_min(1e-6)
    target_dir_unit = target_dir / target_dir_norm.unsqueeze(-1)
    forward_align = torch.sum(x_body * target_dir_unit, dim=-1)
    align_reward = torch.clamp(forward_align, -1.0, 1.0)

    safe_lidar = min_lidar_dist.clamp_min(1e-3)
    obstacle_penalty = 2.0 * torch.exp(-3.0 * safe_lidar) + 0.5 / safe_lidar + torch.relu(0.8 - min_lidar_dist) ** 2 * 20.0

    reward = (
        8.0 * progress_reward
        + 2.0 * dist_reward
        + near_goal_bonus
        + goal_bonus
        + 1.5 * upright_reward
        + 0.8 * align_reward
        - 1.5 * height_penalty
        - 0.6 * speed_penalty
        - 0.2 * smooth_penalty
        - 3.0 * obstacle_penalty
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    return reward
