def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    progress = prev_dist - curr_dist
    r_progress = torch.tanh(4.0 * progress)

    r_dist = torch.exp(-0.25 * curr_dist)
    r_near_goal = torch.exp(-1.2 * curr_dist)
    r_success = (curr_dist < 0.75).float() * 80.0

    z_err = torch.abs(z_pos - 2.0)
    r_height = -2.0 * torch.square(torch.relu(z_err - 0.35))

    lidar = min_lidar_dist.clamp_min(1e-6)
    r_obstacle_soft = -0.8 * torch.exp(-8.0 * lidar) - 0.15 / torch.square(lidar)
    r_obstacle_hard = -(lidar < 0.5).float() * (4.0 / torch.square(lidar) + 20.0 / lidar)

    r_speed = -0.12 * torch.square(torch.relu(v_norm - 4.5))
    r_smooth = -0.05 * torch.clamp(action_diff, min=0.0, max=100.0)
    r_omega = -0.01 * torch.sum(torch.square(omega), dim=-1)

    speed_safe = v_norm.clamp_min(1e-6)
    move_dir = v / speed_safe.unsqueeze(-1)
    align = torch.sum(x_body * move_dir, dim=-1)
    r_align = 0.6 * align * torch.tanh(v_norm)

    r_upright = 1.0 * torch.clamp(1.0 - torch.abs(torch.sum(x_body * torch.tensor([0.0, 0.0, 1.0], device=x_body.device, dtype=x_body.dtype), dim=-1)), min=0.0)

    reward = (
        12.0 * r_progress
        + 3.0 * r_dist
        + 2.0 * r_near_goal
        + r_success
        + r_height
        + r_obstacle_soft
        + r_obstacle_hard
        + r_speed
        + r_smooth
        + r_omega
        + r_align
        + r_upright
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    reward = reward.view(-1)
    return reward
