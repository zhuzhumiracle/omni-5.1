def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    tilt_cos = kwargs["tilt_cos"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    eps = 1e-6

    to_target = target_pos - curr_pos
    target_dist_vec = torch.norm(to_target, dim=-1).clamp_min(eps)
    dir_to_target = to_target / target_dist_vec.unsqueeze(-1)

    vel_norm = v_norm.clamp_min(0.0)
    vel_dir = v / torch.norm(v, dim=-1).clamp_min(eps).unsqueeze(-1)
    align_term = (x_body * dir_to_target).sum(dim=-1).clamp(-1.0, 1.0)
    motion_align = (vel_dir * dir_to_target).sum(dim=-1).clamp(-1.0, 1.0)

    progress = (prev_dist - curr_dist).clamp(-1.0, 1.0)
    progress_reward = 2.0 * progress
    distance_reward = 1.5 * torch.exp(-0.8 * curr_dist)

    reach_reward = 8.0 * torch.exp(-5.0 * curr_dist)

    height_error = (z_pos - 2.0).abs()
    height_reward = -0.8 * torch.tanh(height_error)

    obstacle_soft = -1.5 * torch.exp(-3.0 * min_lidar_dist)
    obstacle_severe = -4.0 / (min_lidar_dist.clamp_min(eps) ** 2)

    speed_penalty = -0.3 * torch.tanh(vel_norm / 4.0)
    smooth_penalty = -0.2 * torch.tanh(action_diff)

    attitude_reward = 0.8 * torch.clamp(tilt_cos, -1.0, 1.0)
    heading_reward = 0.5 * torch.clamp(align_term, -1.0, 1.0)
    motion_reward = 0.3 * torch.clamp(motion_align, -1.0, 1.0)

    omega_penalty = -0.05 * torch.tanh(torch.norm(omega, dim=-1))

    reward = (
        progress_reward
        + distance_reward
        + reach_reward
        + height_reward
        + obstacle_soft
        + obstacle_severe
        + speed_penalty
        + smooth_penalty
        + attitude_reward
        + heading_reward
        + motion_reward
        + omega_penalty
    )

    return reward
