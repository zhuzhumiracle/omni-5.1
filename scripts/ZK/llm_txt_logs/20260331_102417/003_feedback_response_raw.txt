def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    tilt_cos = kwargs.get("tilt_cos", torch.ones_like(curr_dist))
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    eps = 1e-6

    progress = prev_dist - curr_dist
    progress_term = 25.0 * torch.tanh(2.0 * progress)

    target_err = curr_dist
    proximity_term = 8.0 * torch.exp(-0.35 * target_err) - 1.5 * torch.tanh(target_err / 8.0)

    success_bonus = (curr_dist < 0.75).to(curr_dist.dtype) * (300.0 + 700.0 * torch.clamp(0.75 - curr_dist, min=0.0) / 0.75)

    target_vec = target_pos - curr_pos
    target_dir = target_vec / target_vec.norm(dim=-1, keepdim=True).clamp_min(eps)
    vel_dir = v / v_norm.unsqueeze(-1).clamp_min(eps)

    align_target = (x_body * target_dir).sum(dim=-1)
    align_motion = (vel_dir * target_dir).sum(dim=-1)
    alignment_term = 3.0 * torch.tanh(2.5 * align_target) + 2.0 * torch.tanh(2.0 * align_motion)

    upright_term = 1.5 * torch.clamp(tilt_cos, -1.0, 1.0)

    altitude_err = torch.abs(z_pos - 2.0)
    altitude_term = -3.0 * torch.tanh(altitude_err / 0.3)

    speed_penalty = -0.6 * torch.square(torch.relu(v_norm - 4.5))
    low_speed_penalty = -0.3 * torch.relu(0.2 - v_norm)

    omega_norm = torch.norm(omega, dim=-1)
    smooth_term = -0.08 * torch.clamp(action_diff, 0.0, 100.0) - 0.04 * torch.clamp(omega_norm, 0.0, 100.0)

    safe_dist = torch.clamp(min_lidar_dist, min=0.0)
    obstacle_soft = -10.0 * torch.exp(-3.0 * safe_dist) - 40.0 * torch.exp(-10.0 * safe_dist) - 150.0 * torch.exp(-30.0 * safe_dist)
    obstacle_hard = -300.0 * torch.relu(0.45 - safe_dist).pow(2) - 1200.0 * torch.relu(0.25 - safe_dist).pow(2) - 5000.0 * torch.relu(0.10 - safe_dist).pow(2)
    collision_penalty = -20000.0 * (safe_dist < 0.03).to(curr_dist.dtype)

    reward = (
        progress_term
        + proximity_term
        + success_bonus
        + alignment_term
        + upright_term
        + altitude_term
        + speed_penalty
        + low_speed_penalty
        + smooth_term
        + obstacle_soft
        + obstacle_hard
        + collision_penalty
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    reward = reward.reshape(-1)
    return reward
