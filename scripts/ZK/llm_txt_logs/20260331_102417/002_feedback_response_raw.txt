def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    tilt_cos = kwargs["tilt_cos"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    progress = prev_dist - curr_dist
    progress_term = 12.0 * torch.tanh(3.0 * progress)

    proximity_term = 6.0 * torch.tanh((6.0 - curr_dist) / 2.0) - 3.0 * torch.tanh(curr_dist / 10.0)

    success_bonus = (curr_dist < 0.75).to(curr_dist.dtype) * (100.0 + 50.0 * torch.clamp(0.75 - curr_dist, 0.0, 0.75) / 0.75)

    target_vec = target_pos - curr_pos
    target_dir = target_vec / target_vec.norm(dim=-1, keepdim=True).clamp_min(1e-6)
    move_dir = v / v_norm.unsqueeze(-1).clamp_min(1e-6)
    align_target = (x_body * target_dir).sum(dim=-1)
    align_motion = (move_dir * target_dir).sum(dim=-1)
    align_term = 1.5 * torch.tanh(2.0 * align_target) + 0.8 * torch.tanh(2.0 * align_motion)

    upright_term = 1.2 * torch.clamp(tilt_cos, -1.0, 1.0)

    altitude_err = torch.abs(z_pos - 2.0)
    altitude_term = -2.0 * torch.tanh(altitude_err / 0.35)

    speed_over = torch.relu(v_norm - 4.0)
    speed_term = -1.0 * torch.square(speed_over)

    smooth_term = -0.06 * torch.clamp(action_diff, 0.0, 50.0) - 0.03 * torch.clamp(torch.norm(omega, dim=-1), 0.0, 50.0)

    safe_dist = torch.clamp(min_lidar_dist, min=0.0)
    obstacle_soft = -4.0 * torch.exp(-2.5 * safe_dist) - 12.0 * torch.exp(-8.0 * safe_dist)
    obstacle_hard = -80.0 * torch.relu(0.35 - safe_dist).pow(2) - 300.0 * torch.relu(0.15 - safe_dist).pow(2) - 1000.0 * torch.relu(0.05 - safe_dist).pow(2)
    collision_penalty = -2500.0 * (safe_dist < 0.02).to(curr_dist.dtype)

    reward = (
        progress_term
        + proximity_term
        + success_bonus
        + align_term
        + upright_term
        + altitude_term
        + speed_term
        + smooth_term
        + obstacle_soft
        + obstacle_hard
        + collision_penalty
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    reward = reward.reshape(-1)
    return reward
