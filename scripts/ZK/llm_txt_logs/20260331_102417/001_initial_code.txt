def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    tilt_cos = kwargs["tilt_cos"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    progress = prev_dist - curr_dist
    progress_term = 10.0 * torch.tanh(2.0 * progress)

    dist_term = -0.25 * torch.tanh(curr_dist / 8.0)

    reach_bonus = (curr_dist < 0.8).to(curr_dist.dtype) * (20.0 * (1.0 - torch.clamp(curr_dist / 0.8, 0.0, 1.0)))

    target_vec = target_pos - curr_pos
    target_dir = target_vec / target_vec.norm(dim=-1, keepdim=True).clamp_min(1e-6)
    move_dir = v / v_norm.unsqueeze(-1).clamp_min(1e-6)
    align_term = 0.8 * (x_body * target_dir).sum(dim=-1) + 0.4 * (move_dir * target_dir).sum(dim=-1)
    align_term = torch.clamp(align_term, -1.0, 1.0)

    upright_term = 1.5 * tilt_cos

    altitude_err = torch.abs(z_pos - 2.0)
    altitude_term = -1.5 * torch.tanh(altitude_err / 0.4)

    speed_excess = torch.relu(v_norm - 4.0)
    speed_term = -0.8 * torch.square(speed_excess)

    smooth_term = -0.08 * action_diff - 0.02 * torch.norm(omega, dim=-1)

    obstacle_term = -4.0 * torch.exp(-3.0 * min_lidar_dist) - 12.0 * torch.relu(0.45 - min_lidar_dist).pow(2)

    near_collision_penalty = -50.0 * (min_lidar_dist < 0.15).to(curr_dist.dtype)

    reward = (
        progress_term
        + dist_term
        + reach_bonus
        + align_term
        + upright_term
        + altitude_term
        + speed_term
        + smooth_term
        + obstacle_term
        + near_collision_penalty
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    reward = reward.reshape(-1)
    return reward
