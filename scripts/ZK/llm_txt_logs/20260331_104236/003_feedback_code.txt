def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    tilt_cos = kwargs["tilt_cos"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    target_vec = target_pos - curr_pos
    target_dist = torch.norm(target_vec, dim=-1).clamp_min(1e-6)
    target_dir = target_vec / target_dist.unsqueeze(-1)

    vel_norm_safe = v_norm.clamp_min(1e-6)
    vel_dir = v / vel_norm_safe.unsqueeze(-1)

    progress = prev_dist - curr_dist
    progress_reward = 30.0 * torch.tanh(3.0 * progress)

    proximity_reward = 8.0 * torch.exp(-0.9 * curr_dist)
    close_goal_bonus = 120.0 * (curr_dist < 0.5).float()
    near_goal_bonus = 30.0 * (curr_dist < 1.0).float()

    align_vel = (vel_dir * target_dir).sum(dim=-1)
    align_body = (x_body * target_dir).sum(dim=-1)
    alignment_reward = 1.5 * torch.tanh(2.0 * align_vel) + 0.75 * torch.tanh(2.0 * align_body)

    altitude_error = torch.abs(z_pos - 2.0)
    altitude_reward = 2.0 * torch.exp(-1.5 * altitude_error)
    altitude_penalty = 3.0 * torch.relu(altitude_error - 0.4)

    speed_penalty = 0.05 * v_norm.pow(2) + 0.3 * torch.relu(v_norm - 4.0).pow(2)
    smooth_penalty = 0.08 * action_diff + 0.04 * torch.norm(omega, dim=-1)

    clearance = min_lidar_dist.clamp_min(1e-6)
    obstacle_core = 2.5 / clearance.pow(2)
    obstacle_extreme = 0.15 / clearance.pow(4)
    obstacle_penalty = obstacle_core + obstacle_extreme
    obstacle_penalty = torch.clamp(obstacle_penalty, max=1e6)

    collision_penalty = 300.0 * (min_lidar_dist < 0.35).float() + 1000.0 * (min_lidar_dist < 0.2).float()
    near_collision_penalty = 25.0 * torch.exp(-8.0 * min_lidar_dist)

    upright_reward = 1.0 * torch.clamp(tilt_cos, min=-1.0, max=1.0)
    hover_penalty = 0.03 * torch.relu(v_norm - 6.0)

    reward = (
        progress_reward
        + proximity_reward
        + close_goal_bonus
        + near_goal_bonus
        + alignment_reward
        + altitude_reward
        + upright_reward
        - altitude_penalty
        - speed_penalty
        - smooth_penalty
        - obstacle_penalty
        - collision_penalty
        - near_collision_penalty
        - hover_penalty
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    return reward
