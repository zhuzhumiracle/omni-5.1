def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]
    tilt_cos = kwargs.get("tilt_cos", None)

    target_vec = target_pos - curr_pos
    target_dist = torch.norm(target_vec, dim=-1).clamp_min(1e-6)
    target_dir = target_vec / target_dist.unsqueeze(-1)

    vel_norm_safe = v_norm.clamp_min(1e-6)
    vel_dir = v / vel_norm_safe.unsqueeze(-1)

    progress = prev_dist - curr_dist
    progress_reward = 40.0 * torch.tanh(4.0 * progress)
    distance_reward = 12.0 * torch.exp(-0.7 * curr_dist)

    align_vel = (vel_dir * target_dir).sum(dim=-1)
    align_body = (x_body * target_dir).sum(dim=-1)
    alignment_reward = 2.0 * torch.tanh(2.5 * align_vel) + 1.0 * torch.tanh(2.0 * align_body)

    goal_bonus = 500.0 * (curr_dist < 0.5).float() + 100.0 * (curr_dist < 1.0).float()

    altitude_error = torch.abs(z_pos - 2.0)
    altitude_reward = 2.5 * torch.exp(-1.2 * altitude_error)
    altitude_penalty = 4.0 * torch.relu(altitude_error - 0.35)

    speed_penalty = 0.06 * v_norm.pow(2) + 0.4 * torch.relu(v_norm - 4.5).pow(2)
    smooth_penalty = 0.08 * action_diff + 0.03 * torch.norm(omega, dim=-1)

    clearance = min_lidar_dist.clamp_min(1e-6)
    obstacle_soft = 3.0 * torch.exp(-6.0 * clearance)
    obstacle_hard = 0.08 / clearance.pow(2) + 0.002 / clearance.pow(4)
    obstacle_penalty = obstacle_soft + obstacle_hard
    obstacle_penalty = torch.clamp(obstacle_penalty, max=1e5)

    collision_penalty = 0.0
    collision_penalty = collision_penalty + 500.0 * (min_lidar_dist < 0.5).float()
    collision_penalty = collision_penalty + 2000.0 * (min_lidar_dist < 0.3).float()
    collision_penalty = collision_penalty + 8000.0 * (min_lidar_dist < 0.15).float()
    collision_penalty = collision_penalty + 20000.0 * (min_lidar_dist < 0.08).float()

    near_collision_penalty = 40.0 * torch.exp(-10.0 * min_lidar_dist)

    if tilt_cos is None:
        upright_reward = torch.zeros_like(curr_dist)
    else:
        upright_reward = 1.5 * torch.clamp(tilt_cos, min=-1.0, max=1.0)

    reward = (
        progress_reward
        + distance_reward
        + alignment_reward
        + goal_bonus
        + altitude_reward
        + upright_reward
        - altitude_penalty
        - speed_penalty
        - smooth_penalty
        - obstacle_penalty
        - collision_penalty
        - near_collision_penalty
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    return reward
