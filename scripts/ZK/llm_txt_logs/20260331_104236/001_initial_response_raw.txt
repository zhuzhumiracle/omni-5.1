def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    tilt_cos = kwargs["tilt_cos"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    progress = prev_dist - curr_dist
    progress_reward = 8.0 * torch.tanh(3.0 * progress)

    dist_reward = 2.0 * torch.exp(-0.7 * curr_dist)
    close_goal_bonus = 12.0 * (curr_dist < 0.5).float() * torch.exp(-4.0 * curr_dist)

    target_vec = target_pos - curr_pos
    target_dir = target_vec / target_vec.norm(dim=-1, keepdim=True).clamp_min(1e-6)
    vel_dir = v / v_norm.unsqueeze(-1).clamp_min(1e-6)
    align_vel = (vel_dir * target_dir).sum(dim=-1)
    align_body = (x_body * target_dir).sum(dim=-1)
    align_reward = 0.8 * torch.tanh(align_vel) + 0.4 * torch.tanh(align_body)

    z_err = torch.abs(z_pos - 2.0)
    altitude_reward = 1.2 * torch.exp(-1.5 * z_err) - 0.8 * z_err

    speed_penalty = 0.15 * torch.relu(v_norm - 4.0).pow(2) + 0.03 * v_norm.pow(2)
    smooth_penalty = 0.08 * action_diff + 0.02 * torch.norm(omega, dim=-1)

    obstacle_penalty = 3.0 / torch.clamp(min_lidar_dist, min=1e-3).pow(2)
    obstacle_penalty = torch.clamp(obstacle_penalty, max=1e4)

    upright_reward = 1.5 * tilt_cos

    reward = (
        progress_reward
        + dist_reward
        + close_goal_bonus
        + align_reward
        + altitude_reward
        + upright_reward
        - speed_penalty
        - smooth_penalty
        - obstacle_penalty
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    return reward
