def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    tilt_cos = kwargs["tilt_cos"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    dist_progress = prev_dist - curr_dist
    dist_progress_term = torch.tanh(2.0 * dist_progress)

    target_reach_term = torch.exp(-2.5 * curr_dist) * (curr_dist < 1.0).float() * 8.0
    near_goal_bonus = (curr_dist < 0.5).float() * 20.0

    altitude_error = torch.abs(z_pos - 2.0)
    altitude_term = -torch.tanh(1.5 * altitude_error) - 0.5 * torch.relu(altitude_error - 0.5)

    obstacle_term = -3.0 * torch.exp(-4.0 * min_lidar_dist.clamp_min(0.0))
    severe_obstacle_penalty = -8.0 * torch.relu(0.6 - min_lidar_dist).pow(2) - 20.0 * torch.relu(0.25 - min_lidar_dist).pow(2)

    speed_penalty = -0.5 * torch.relu(v_norm - 4.0).pow(2) - 0.1 * torch.relu(v_norm - 6.0)
    smooth_penalty = -0.15 * action_diff - 0.05 * torch.norm(omega, dim=-1)

    upright_term = 0.8 * tilt_cos.clamp(-1.0, 1.0)

    target_vec = target_pos - curr_pos
    target_dir = target_vec / torch.norm(target_vec, dim=-1, keepdim=True).clamp_min(1e-6)
    vel_dir = v / v_norm.unsqueeze(-1).clamp_min(1e-6)
    align_term = 0.5 * (x_body * target_dir).sum(dim=-1) + 0.25 * (vel_dir * target_dir).sum(dim=-1)

    moving_bonus = 0.2 * torch.tanh(v_norm)

    reward = (
        4.0 * dist_progress_term
        + 2.5 * target_reach_term
        + near_goal_bonus
        + altitude_term
        + obstacle_term
        + severe_obstacle_penalty
        + speed_penalty
        + smooth_penalty
        + upright_term
        + align_term
        + moving_bonus
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    return reward
