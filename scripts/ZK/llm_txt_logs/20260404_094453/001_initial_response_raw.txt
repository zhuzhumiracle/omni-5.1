def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    tilt_cos = kwargs["tilt_cos"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    eps = 1e-6

    target_vec = target_pos - curr_pos
    target_dir = target_vec / torch.norm(target_vec, dim=-1).clamp_min(eps).unsqueeze(-1)
    forward_align = (x_body * target_dir).sum(dim=-1).clamp(-1.0, 1.0)

    omega_norm = torch.norm(omega, dim=-1)
    v_xy_norm = torch.norm(v[:, :2], dim=-1)
    v_z_abs = torch.abs(v[:, 2])

    progress = torch.tanh(2.5 * (prev_dist - curr_dist))
    close_to_goal = torch.exp(-0.8 * curr_dist)
    goal_bonus = (curr_dist < 0.75).float() * 18.0 + (curr_dist < 0.35).float() * 35.0

    forward_reward = 0.8 * torch.relu(forward_align) * torch.tanh(v_norm / 2.5)

    speed_target = 2.0
    speed_band_penalty = 0.45 * torch.square(v_norm - speed_target)
    low_speed_penalty = 0.6 * torch.relu(1.0 - v_norm).square()
    high_speed_penalty = 0.25 * torch.relu(v_norm - 4.0).square()

    global_tilt_penalty = 0.8 * torch.relu(0.92 - tilt_cos) + 0.4 * torch.square(torch.relu(0.98 - tilt_cos))

    altitude_reg = 0.35 * torch.square(z_pos - 2.0)

    smooth_penalty = 0.12 * action_diff + 0.08 * omega_norm + 0.03 * torch.square(omega_norm)

    safe_dist = min_lidar_dist.clamp_min(eps)
    inv_dist = 1.0 / safe_dist

    near_mask = torch.sigmoid((1.8 - min_lidar_dist) * 3.0)
    critical_mask = torch.sigmoid((0.55 - min_lidar_dist) * 10.0)

    obstacle_penalty = (
        0.25 * torch.exp(-1.2 * safe_dist)
        + 0.9 * torch.exp(-3.5 * safe_dist)
        + 1.8 * torch.square(torch.relu(0.45 - safe_dist)) * inv_dist
    )

    near_speed_penalty = near_mask * (
        0.18 * v_norm + 0.12 * torch.square(v_norm) + 0.10 * v_xy_norm
    )
    near_vz_penalty = near_mask * (
        0.28 * v_z_abs + 0.18 * torch.square(v_z_abs)
    )
    near_omega_penalty = near_mask * (
        0.10 * omega_norm + 0.08 * torch.square(omega_norm)
    )

    upward_escape_penalty = near_mask * torch.relu(v[:, 2]) * 0.35
    climbing_bias_penalty = near_mask * torch.relu(z_pos - 2.4) * 0.12

    death_like = (
        (tilt_cos < 0.18).float()
        + (z_pos < 0.8).float()
        + (z_pos > 3.5).float()
    ).clamp(max=1.0) * 50.0

    reward = (
        4.5 * progress
        + 2.0 * close_to_goal
        + goal_bonus
        + forward_reward
        - speed_band_penalty
        - low_speed_penalty
        - high_speed_penalty
        - global_tilt_penalty
        - altitude_reg
        - smooth_penalty
        - obstacle_penalty
        - near_speed_penalty
        - near_vz_penalty
        - near_omega_penalty
        - upward_escape_penalty
        - climbing_bias_penalty
        - critical_mask * 8.0
        - death_like
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = reward.reshape(-1)
    return reward
