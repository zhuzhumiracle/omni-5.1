def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    eps = 1e-6

    target_vec = target_pos - curr_pos
    target_dist = torch.norm(target_vec, dim=-1).clamp_min(eps)
    target_dir = target_vec / target_dist.unsqueeze(-1)

    v_dir = v / v_norm.unsqueeze(-1).clamp_min(eps)
    x_body_norm = torch.norm(x_body, dim=-1).clamp_min(eps)
    x_body_dir = x_body / x_body_norm.unsqueeze(-1)

    progress = prev_dist - curr_dist
    r_progress = torch.tanh(12.0 * progress)

    r_close = torch.exp(-0.25 * curr_dist)
    r_goal_near = torch.exp(-1.0 * curr_dist)

    align_v = (v_dir * target_dir).sum(dim=-1)
    align_body = (x_body_dir * target_dir).sum(dim=-1)
    r_align = 0.5 * (align_v + align_body)

    alt_err = torch.abs(z_pos - 2.0)
    r_alt = torch.exp(-1.2 * alt_err)
    r_alt_pen = torch.relu(alt_err - 0.4)

    omega_norm = torch.norm(omega, dim=-1)
    r_turn = torch.tanh(0.4 * omega_norm)
    r_smooth = torch.tanh(action_diff)

    speed_high = torch.relu(v_norm - 6.0)
    speed_low = torch.relu(0.15 - v_norm)
    r_speed_pen = speed_high.pow(2) + 0.2 * speed_low.pow(2)

    d = min_lidar_dist.clamp_min(eps)

    inv_d = 1.0 / d
    inv_d2 = inv_d * inv_d
    inv_d4 = inv_d2 * inv_d2

    very_close = torch.relu(0.7 - d)
    danger = torch.relu(0.35 - d)
    collision = torch.relu(0.15 - d)

    obstacle_pen = (
        15.0 * torch.exp(-2.5 * d)
        + 60.0 * torch.exp(-8.0 * d)
        + 250.0 * torch.exp(-20.0 * d)
        + 1000.0 * torch.exp(-50.0 * d)
        + 4000.0 * torch.exp(-120.0 * d)
        + 20000.0 * torch.exp(-300.0 * d)
        + 0.5 * very_close.pow(2)
        + 8.0 * danger.pow(2)
        + 80.0 * collision.pow(2)
        + 2.0 * inv_d2 * (d < 0.5).to(d.dtype)
        + 15.0 * inv_d4 * (d < 0.25).to(d.dtype)
    )

    crash_penalty = torch.sigmoid((0.2 - d) * 50.0) * 50000.0

    success_bonus = (curr_dist < 0.5).to(curr_dist.dtype) * 8000.0
    reach_bonus = (curr_dist < 1.5).to(curr_dist.dtype) * 1500.0

    reward = (
        80.0 * r_progress
        + 25.0 * r_close
        + 10.0 * r_goal_near
        + 6.0 * r_align
        + 4.0 * r_alt
        + reach_bonus
        + success_bonus
        - 3.0 * r_speed_pen
        - 1.0 * r_turn
        - 0.8 * r_smooth
        - 3.0 * r_alt_pen
        - 20.0 * obstacle_pen
        - crash_penalty
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    return reward
