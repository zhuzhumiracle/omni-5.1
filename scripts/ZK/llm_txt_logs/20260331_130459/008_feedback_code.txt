def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    eps = 1e-6

    target_vec = target_pos - curr_pos
    target_dir = target_vec / torch.norm(target_vec, dim=-1, keepdim=True).clamp_min(eps)
    v_dir = v / v_norm.unsqueeze(-1).clamp_min(eps)
    x_body_dir = x_body / torch.norm(x_body, dim=-1, keepdim=True).clamp_min(eps)

    progress = prev_dist - curr_dist
    r_progress = torch.tanh(10.0 * progress)

    r_close = torch.exp(-0.35 * curr_dist)
    r_near = torch.exp(-0.9 * curr_dist)
    r_goal_bonus = (curr_dist < 0.75).to(curr_dist.dtype) * 1000.0
    r_success_bonus = (curr_dist < 0.25).to(curr_dist.dtype) * 4000.0

    align_v = (v_dir * target_dir).sum(dim=-1)
    align_body = (x_body_dir * target_dir).sum(dim=-1)
    r_align = 0.5 * align_v + 0.5 * align_body

    alt_err = torch.abs(z_pos - 2.0)
    r_alt = torch.exp(-1.5 * alt_err)
    r_alt_pen = torch.relu(alt_err - 0.35)

    omega_norm = torch.norm(omega, dim=-1)
    r_smooth = torch.tanh(action_diff)
    r_turn = torch.tanh(0.5 * omega_norm)

    speed_over = torch.relu(v_norm - 6.0)
    speed_under = torch.relu(0.25 - v_norm)
    speed_pen = speed_over.pow(2) + 0.5 * speed_under.pow(2)

    d = min_lidar_dist.clamp_min(eps)
    inv_d = 1.0 / d

    obstacle_hard = (
        8.0 * torch.exp(-2.0 * d)
        + 25.0 * torch.exp(-8.0 * d)
        + 80.0 * torch.exp(-20.0 * d)
        + 300.0 * torch.exp(-50.0 * d)
        + 1200.0 * torch.exp(-120.0 * d)
        + 5000.0 * torch.exp(-300.0 * d)
    )

    collision_zone = (d < 0.6).to(d.dtype)
    danger_zone = (d < 0.3).to(d.dtype)
    critical_zone = (d < 0.12).to(d.dtype)

    obstacle_collision = collision_zone * (
        30.0 * torch.relu(0.6 - d).pow(2)
        + 150.0 * inv_d
        + 400.0 * inv_d.pow(2)
    )
    obstacle_danger = danger_zone * (
        200.0
        + 500.0 * torch.relu(0.3 - d)
        + 1500.0 * inv_d.pow(2)
        + 5000.0 * inv_d.pow(3)
    )
    obstacle_critical = critical_zone * (
        2000.0
        + 2500.0 * torch.relu(0.12 - d)
        + 20000.0 * inv_d.pow(3)
        + 50000.0 * inv_d.pow(4)
    )

    reward = (
        35.0 * r_progress
        + 18.0 * r_close
        + 28.0 * r_near
        + 3.0 * r_align
        + 2.5 * r_alt
        + r_goal_bonus
        + r_success_bonus
        - 1.5 * speed_pen
        - 0.8 * r_turn
        - 0.25 * r_smooth
        - 1.5 * r_alt_pen
        - 20.0 * obstacle_hard
        - obstacle_collision
        - obstacle_danger
        - obstacle_critical
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    return reward
