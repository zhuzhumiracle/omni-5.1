def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]
    tilt_cos = kwargs["tilt_cos"]

    progress = prev_dist - curr_dist
    progress_reward = torch.tanh(8.0 * progress)

    goal_close = torch.exp(-0.8 * curr_dist)
    goal_bonus = (curr_dist < 1.0).to(curr_dist.dtype) * 25.0
    goal_reaching_reward = 8.0 * goal_close + goal_bonus

    target_dir = target_pos - curr_pos
    target_dir_norm = torch.norm(target_dir, dim=-1).clamp_min(1e-6)
    target_dir_unit = target_dir / target_dir_norm.unsqueeze(-1)

    v_norm_safe = v_norm.clamp_min(1e-6)
    v_dir = v / v_norm_safe.unsqueeze(-1)
    x_body_norm = torch.norm(x_body, dim=-1).clamp_min(1e-6)
    x_body_unit = x_body / x_body_norm.unsqueeze(-1)

    vel_align = (v_dir * target_dir_unit).sum(dim=-1)
    body_align = (x_body_unit * target_dir_unit).sum(dim=-1)
    alignment_reward = 0.5 * vel_align + 0.5 * body_align

    altitude_error = torch.abs(z_pos - 2.0)
    altitude_reward = torch.exp(-1.5 * altitude_error) - 1.0

    speed_penalty = torch.relu(v_norm - 5.0).pow(2)
    turn_penalty = torch.tanh(torch.norm(omega, dim=-1))
    smooth_penalty = torch.tanh(action_diff)

    obstacle_safe = min_lidar_dist.clamp_min(1e-6)
    obstacle_hard_penalty = torch.where(
        min_lidar_dist < 0.6,
        1.0 / obstacle_safe.pow(2) + 5.0 / obstacle_safe,
        torch.zeros_like(min_lidar_dist),
    )
    obstacle_soft_penalty = torch.exp(-2.5 * min_lidar_dist)

    upright_reward = (tilt_cos.clamp(-1.0, 1.0) + 1.0) * 0.5

    reward = (
        12.0 * progress_reward
        + goal_reaching_reward
        + 2.5 * alignment_reward
        + 1.5 * altitude_reward
        + 1.0 * upright_reward
        - 0.3 * speed_penalty
        - 0.2 * turn_penalty
        - 0.15 * smooth_penalty
        - 3.0 * obstacle_soft_penalty
        - 0.25 * obstacle_hard_penalty
    )

    collision_severe = (min_lidar_dist < 0.25).to(min_lidar_dist.dtype) * (
        -50.0 - 200.0 * (0.25 - min_lidar_dist).clamp_min(0.0)
    )
    near_collision = (min_lidar_dist < 0.5).to(min_lidar_dist.dtype) * (
        -10.0 * (0.5 - min_lidar_dist).clamp_min(0.0).pow(2)
    )

    reward = reward + collision_severe + near_collision
    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    return reward
