def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    tilt_cos = kwargs["tilt_cos"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    progress = prev_dist - curr_dist
    progress_reward = torch.tanh(3.0 * progress)

    reach_bonus = torch.exp(-2.5 * curr_dist)
    near_goal_bonus = (curr_dist < 1.0).to(curr_dist.dtype) * torch.exp(-1.5 * curr_dist) * 5.0

    target_dir = target_pos - curr_pos
    target_dir_norm = torch.norm(target_dir, dim=-1).clamp_min(1e-6)
    target_dir_unit = target_dir / target_dir_norm.unsqueeze(-1)
    v_dir_norm = torch.norm(v, dim=-1).clamp_min(1e-6)
    v_dir_unit = v / v_dir_norm.unsqueeze(-1)
    body_align = (x_body * target_dir_unit).sum(dim=-1)
    vel_align = (v_dir_unit * target_dir_unit).sum(dim=-1)
    align_reward = 0.5 * body_align + 0.5 * vel_align

    z_error = torch.abs(z_pos - 2.0)
    altitude_reward = -torch.tanh(1.5 * z_error)

    speed_penalty = torch.relu(v_norm - 4.0).pow(2)
    smooth_penalty = torch.tanh(0.5 * action_diff) + 0.25 * torch.tanh(0.5 * torch.norm(omega, dim=-1))
    upright_reward = torch.clamp(tilt_cos, min=-1.0, max=1.0)

    safe_dist = min_lidar_dist.clamp_min(1e-6)
    obstacle_penalty = 1.5 / safe_dist.pow(2)
    close_obstacle_penalty = torch.relu(0.8 - min_lidar_dist).pow(2) * 10.0

    reward = (
        3.0 * progress_reward
        + 2.0 * reach_bonus
        + near_goal_bonus
        + 0.8 * align_reward
        + 0.8 * altitude_reward
        + 1.2 * upright_reward
        - 0.35 * speed_penalty
        - 0.25 * smooth_penalty
        - obstacle_penalty
        - close_obstacle_penalty
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    return reward
