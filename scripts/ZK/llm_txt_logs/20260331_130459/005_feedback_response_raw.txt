def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    target_vec = target_pos - curr_pos
    target_vec_norm = torch.norm(target_vec, dim=-1).clamp_min(1e-6)
    target_dir = target_vec / target_vec_norm.unsqueeze(-1)

    v_norm_safe = v_norm.clamp_min(1e-6)
    v_dir = v / v_norm_safe.unsqueeze(-1)

    x_body_norm = torch.norm(x_body, dim=-1).clamp_min(1e-6)
    x_body_dir = x_body / x_body_norm.unsqueeze(-1)

    progress = prev_dist - curr_dist
    progress_reward = torch.tanh(20.0 * progress)

    proximity_reward = torch.exp(-0.35 * curr_dist)
    goal_bonus = (curr_dist < 0.8).to(curr_dist.dtype) * 800.0
    success_bonus = (curr_dist < 0.4).to(curr_dist.dtype) * 4000.0

    vel_align = (v_dir * target_dir).sum(dim=-1)
    body_align = (x_body_dir * target_dir).sum(dim=-1)
    alignment_reward = 0.5 * vel_align + 0.5 * body_align

    altitude_error = torch.abs(z_pos - 2.0)
    altitude_reward = torch.exp(-1.2 * altitude_error)

    omega_norm = torch.norm(omega, dim=-1)
    smooth_penalty = torch.tanh(action_diff)
    turn_penalty = torch.tanh(omega_norm)

    speed_penalty = torch.relu(v_norm - 5.0).pow(2)
    low_speed_penalty = torch.relu(0.35 - v_norm).pow(2)

    min_dist = min_lidar_dist.clamp_min(1e-6)
    inv_dist = 1.0 / min_dist

    obstacle_penalty = (
        0.8 * torch.exp(-1.5 * min_dist)
        + 6.0 * torch.exp(-6.0 * min_dist)
        + 40.0 * torch.exp(-18.0 * min_dist)
        + 200.0 * torch.exp(-45.0 * min_dist)
        + 800.0 * torch.exp(-90.0 * min_dist)
        + 2500.0 * torch.exp(-180.0 * min_dist)
    )

    crash_penalty = torch.where(
        min_lidar_dist < 0.35,
        1000.0 * torch.relu(0.35 - min_lidar_dist).pow(2) + 20.0 * inv_dist.pow(2),
        torch.zeros_like(min_lidar_dist),
    )
    severe_crash_penalty = torch.where(
        min_lidar_dist < 0.2,
        5000.0 + 20000.0 * torch.relu(0.2 - min_lidar_dist) + 200.0 * inv_dist.pow(3),
        torch.zeros_like(min_lidar_dist),
    )

    reward = (
        40.0 * progress_reward
        + 12.0 * proximity_reward
        + 4.0 * alignment_reward
        + 2.0 * altitude_reward
        + goal_bonus
        + success_bonus
        - 0.5 * speed_penalty
        - 0.35 * low_speed_penalty
        - 0.15 * turn_penalty
        - 0.08 * smooth_penalty
        - 8.0 * obstacle_penalty
        - crash_penalty
        - severe_crash_penalty
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    return reward
