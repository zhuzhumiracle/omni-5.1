def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    target_vec = target_pos - curr_pos
    target_dir = target_vec / torch.norm(target_vec, dim=-1).clamp_min(1e-6).unsqueeze(-1)

    v_dir = v / v_norm.clamp_min(1e-6).unsqueeze(-1)
    x_body_dir = x_body / torch.norm(x_body, dim=-1).clamp_min(1e-6).unsqueeze(-1)

    progress = prev_dist - curr_dist
    progress_reward = torch.tanh(15.0 * progress)

    proximity_reward = torch.exp(-0.6 * curr_dist)
    near_goal_reward = torch.exp(-2.5 * curr_dist)
    goal_bonus = (curr_dist < 0.75).to(curr_dist.dtype) * 1200.0
    success_bonus = (curr_dist < 0.35).to(curr_dist.dtype) * 6000.0

    vel_align = (v_dir * target_dir).sum(dim=-1)
    body_align = (x_body_dir * target_dir).sum(dim=-1)
    alignment_reward = 0.5 * vel_align + 0.5 * body_align

    altitude_error = torch.abs(z_pos - 2.0)
    altitude_reward = torch.exp(-1.5 * altitude_error)
    altitude_penalty = torch.relu(altitude_error - 0.4)

    omega_norm = torch.norm(omega, dim=-1)
    smooth_penalty = torch.tanh(action_diff)
    turn_penalty = torch.tanh(omega_norm)

    speed_penalty = torch.relu(v_norm - 5.0).pow(2)
    low_speed_penalty = torch.relu(0.25 - v_norm).pow(2)

    min_dist = min_lidar_dist.clamp_min(1e-6)
    inv_dist = 1.0 / min_dist

    obstacle_soft = (
        0.2 * torch.exp(-1.0 * min_dist)
        + 1.0 * torch.exp(-3.0 * min_dist)
        + 5.0 * torch.exp(-10.0 * min_dist)
        + 20.0 * torch.exp(-30.0 * min_dist)
        + 80.0 * torch.exp(-80.0 * min_dist)
        + 250.0 * torch.exp(-160.0 * min_dist)
        + 800.0 * torch.exp(-320.0 * min_dist)
    )

    collision_zone = (min_dist < 0.5).to(min_dist.dtype)
    severe_zone = (min_dist < 0.25).to(min_dist.dtype)
    critical_zone = (min_dist < 0.12).to(min_dist.dtype)

    obstacle_hard = collision_zone * (
        40.0 * torch.relu(0.5 - min_dist).pow(2)
        + 20.0 * inv_dist
        + 200.0 * torch.exp(-12.0 * min_dist)
    )
    obstacle_severe = severe_zone * (
        500.0
        + 200.0 * torch.relu(0.25 - min_dist)
        + 1000.0 * inv_dist.pow(2)
    )
    obstacle_critical = critical_zone * (
        5000.0
        + 2000.0 * torch.relu(0.12 - min_dist)
        + 20000.0 * inv_dist.pow(3)
    )

    reward = (
        25.0 * progress_reward
        + 18.0 * proximity_reward
        + 30.0 * near_goal_reward
        + 5.0 * alignment_reward
        + 2.5 * altitude_reward
        + goal_bonus
        + success_bonus
        - 0.8 * speed_penalty
        - 0.4 * low_speed_penalty
        - 0.25 * turn_penalty
        - 0.12 * smooth_penalty
        - 1.5 * altitude_penalty
        - 12.0 * obstacle_soft
        - obstacle_hard
        - obstacle_severe
        - obstacle_critical
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    return reward
