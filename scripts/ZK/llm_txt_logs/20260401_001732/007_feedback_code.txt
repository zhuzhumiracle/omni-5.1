def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    dist = curr_dist.clamp_min(1e-6)
    prev_dist_safe = prev_dist.clamp_min(1e-6)
    speed = v_norm.clamp_min(1e-6)
    lidar = min_lidar_dist.clamp_min(1e-6)

    v_dir = v / speed.unsqueeze(-1)
    heading_align = torch.sum(x_body * v_dir, dim=-1).clamp(-1.0, 1.0)

    to_goal = target_pos - curr_pos
    to_goal_norm = torch.norm(to_goal, dim=-1).clamp_min(1e-6)
    goal_dir_align = torch.sum((to_goal / to_goal_norm.unsqueeze(-1)) * v_dir, dim=-1).clamp(-1.0, 1.0)

    progress = prev_dist_safe - dist
    progress_reward = 60.0 * torch.tanh(8.0 * progress)

    goal_proximity_reward = 80.0 * torch.exp(-0.8 * dist)
    shaped_goal_reward = 15.0 * torch.exp(-0.2 * dist)

    success_bonus = (dist < 0.75).to(dist.dtype) * 3000.0
    near_goal_bonus = (dist < 2.5).to(dist.dtype) * 250.0

    height_error = torch.abs(z_pos - 2.0)
    height_penalty = -20.0 * torch.square(torch.relu(height_error - 0.3))

    move_reward = 10.0 * torch.tanh(speed) * heading_align
    approach_reward = 8.0 * torch.tanh(speed) * goal_dir_align

    omega_norm = torch.norm(omega, dim=-1)
    smooth_penalty = -0.08 * action_diff - 0.08 * torch.tanh(omega_norm)

    overspeed_penalty = -2.0 * torch.square(torch.relu(speed - 5.0))

    obstacle_near = -30.0 * torch.exp(-6.0 * lidar)
    obstacle_close = -120.0 * torch.exp(-18.0 * lidar)
    obstacle_critical = -600.0 * torch.exp(-60.0 * lidar)
    collision_penalty = -(lidar < 0.45).to(lidar.dtype) * (500.0 + 1500.0 / (lidar + 0.02))
    crash_penalty = -(lidar < 0.2).to(lidar.dtype) * (3000.0 + 8000.0 / (lidar + 0.005))
    obstacle_inverse = -2.0 / (lidar * lidar + 1e-4)

    reward = (
        progress_reward
        + goal_proximity_reward
        + shaped_goal_reward
        + success_bonus
        + near_goal_bonus
        + height_penalty
        + move_reward
        + approach_reward
        + smooth_penalty
        + overspeed_penalty
        + obstacle_near
        + obstacle_close
        + obstacle_critical
        + collision_penalty
        + crash_penalty
        + obstacle_inverse
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    return reward
