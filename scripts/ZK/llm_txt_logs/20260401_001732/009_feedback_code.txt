def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    dist = curr_dist.clamp_min(1e-6)
    prev_dist_safe = prev_dist.clamp_min(1e-6)
    speed = v_norm.clamp_min(1e-6)
    lidar = min_lidar_dist.clamp_min(1e-6)

    v_dir = v / speed.unsqueeze(-1)
    to_goal = target_pos - curr_pos
    to_goal_norm = torch.norm(to_goal, dim=-1).clamp_min(1e-6)
    goal_dir = to_goal / to_goal_norm.unsqueeze(-1)

    body_align = torch.sum(x_body * v_dir, dim=-1).clamp(-1.0, 1.0)
    goal_align = torch.sum(goal_dir * v_dir, dim=-1).clamp(-1.0, 1.0)

    progress = prev_dist_safe - dist
    progress_reward = 80.0 * torch.tanh(6.0 * progress)

    proximity_reward = 250.0 * torch.exp(-0.9 * dist)
    near_goal_bonus = (dist < 1.5).to(dist.dtype) * 400.0
    success_bonus = (dist < 0.6).to(dist.dtype) * 6000.0

    forward_reward = 8.0 * torch.tanh(speed) * body_align
    approach_reward = 14.0 * torch.tanh(speed) * goal_align

    height_error = torch.abs(z_pos - 2.0)
    height_penalty = -30.0 * torch.square(torch.relu(height_error - 0.25))

    omega_norm = torch.norm(omega, dim=-1)
    smooth_penalty = -0.05 * action_diff - 0.08 * torch.tanh(omega_norm)

    overspeed_penalty = -4.0 * torch.square(torch.relu(speed - 5.0))

    obstacle_base = -15.0 * torch.exp(-2.5 * lidar)
    obstacle_mid = -120.0 * torch.exp(-8.0 * lidar)
    obstacle_strong = -600.0 * torch.exp(-20.0 * lidar)
    obstacle_critical = -3500.0 * torch.exp(-60.0 * lidar)
    collision_penalty = -(lidar < 0.35).to(lidar.dtype) * (5000.0 + 8000.0 / (lidar + 0.02))
    crash_penalty = -(lidar < 0.15).to(lidar.dtype) * (12000.0 + 20000.0 / (lidar + 0.005))
    severe_inverse = -30.0 / (lidar * lidar + 1e-4)

    reward = (
        progress_reward
        + proximity_reward
        + near_goal_bonus
        + success_bonus
        + forward_reward
        + approach_reward
        + height_penalty
        + smooth_penalty
        + overspeed_penalty
        + obstacle_base
        + obstacle_mid
        + obstacle_strong
        + obstacle_critical
        + collision_penalty
        + crash_penalty
        + severe_inverse
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    return reward
