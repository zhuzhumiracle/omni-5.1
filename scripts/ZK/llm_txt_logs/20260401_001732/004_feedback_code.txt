def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    dist = curr_dist.clamp_min(1e-6)
    prev_dist_safe = prev_dist.clamp_min(1e-6)
    speed = v_norm.clamp_min(0.0)
    safe_lidar = min_lidar_dist.clamp_min(1e-6)

    progress = prev_dist_safe - dist
    progress_reward = 35.0 * torch.tanh(4.0 * progress)

    proximity_reward = 12.0 * torch.exp(-0.8 * dist)

    success_bonus = (dist < 0.75).to(dist.dtype) * 800.0

    height_error = torch.abs(z_pos - 2.0)
    height_reward = -3.0 * torch.square(torch.relu(height_error - 0.25))

    v_dir = v / speed.unsqueeze(-1).clamp_min(1e-6)
    heading_align = torch.sum(x_body * v_dir, dim=-1)
    move_reward = 2.0 * torch.tanh(speed) * heading_align

    omega_norm = torch.norm(omega, dim=-1)
    smooth_penalty = -0.12 * action_diff - 0.05 * omega_norm

    speed_penalty = -0.35 * torch.square(torch.relu(speed - 5.0))

    obstacle_near = -12.0 * torch.exp(-4.0 * safe_lidar) - 4.0 / (safe_lidar * safe_lidar + 1e-4)
    obstacle_critical = -(safe_lidar < 0.5).to(safe_lidar.dtype) * (40.0 + 160.0 / (safe_lidar + 0.01))
    obstacle_collision = -(safe_lidar < 0.2).to(safe_lidar.dtype) * (300.0 + 700.0 / (safe_lidar + 0.005))

    approach_dir = torch.sum((target_pos - curr_pos) * v_dir, dim=-1)
    approach_reward = 1.0 * torch.tanh(approach_dir / dist.clamp_min(1e-6))

    reward = (
        progress_reward
        + proximity_reward
        + success_bonus
        + height_reward
        + move_reward
        + smooth_penalty
        + speed_penalty
        + obstacle_near
        + obstacle_critical
        + obstacle_collision
        + approach_reward
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    return reward
