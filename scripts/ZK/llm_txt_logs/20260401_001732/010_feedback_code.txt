def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    dist = curr_dist.clamp_min(1e-6)
    prev_dist = prev_dist.clamp_min(1e-6)
    speed = v_norm.clamp_min(1e-6)
    lidar = min_lidar_dist.clamp_min(1e-6)

    to_goal = target_pos - curr_pos
    to_goal_norm = torch.norm(to_goal, dim=-1).clamp_min(1e-6)
    goal_dir = to_goal / to_goal_norm.unsqueeze(-1)

    vel_dir = v / speed.unsqueeze(-1)
    body_align = torch.sum(x_body * vel_dir, dim=-1).clamp(-1.0, 1.0)
    goal_align = torch.sum(goal_dir * vel_dir, dim=-1).clamp(-1.0, 1.0)

    progress = prev_dist - dist
    progress_reward = 120.0 * torch.tanh(8.0 * progress)

    proximity_reward = 60.0 * torch.exp(-0.8 * dist)
    close_bonus = (dist < 2.0).to(dist.dtype) * 150.0
    success_bonus = (dist < 0.5).to(dist.dtype) * 5000.0

    heading_reward = 12.0 * torch.tanh(speed) * body_align
    goal_heading_reward = 18.0 * torch.tanh(speed) * goal_align

    height_error = torch.abs(z_pos - 2.0)
    height_penalty = -25.0 * torch.square(torch.relu(height_error - 0.25))

    omega_norm = torch.norm(omega, dim=-1)
    smooth_penalty = -0.03 * action_diff - 0.05 * torch.tanh(omega_norm)

    overspeed_penalty = -3.0 * torch.square(torch.relu(speed - 5.0))

    obstacle_soft = -8.0 * torch.exp(-2.0 * lidar)
    obstacle_mid = -25.0 * torch.exp(-6.0 * lidar)
    obstacle_strong = -120.0 * torch.exp(-15.0 * lidar)
    obstacle_critical = -800.0 * torch.exp(-40.0 * lidar)
    obstacle_extreme = -5000.0 * torch.exp(-120.0 * lidar)

    collision_penalty = -(lidar < 0.35).to(lidar.dtype) * (3000.0 + 7000.0 / (lidar + 0.02))
    crash_penalty = -(lidar < 0.15).to(lidar.dtype) * (8000.0 + 15000.0 / (lidar + 0.005))
    near_zero_penalty = -100.0 / (lidar * lidar + 1e-4)

    reward = (
        progress_reward
        + proximity_reward
        + close_bonus
        + success_bonus
        + heading_reward
        + goal_heading_reward
        + height_penalty
        + smooth_penalty
        + overspeed_penalty
        + obstacle_soft
        + obstacle_mid
        + obstacle_strong
        + obstacle_critical
        + obstacle_extreme
        + collision_penalty
        + crash_penalty
        + near_zero_penalty
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    return reward
