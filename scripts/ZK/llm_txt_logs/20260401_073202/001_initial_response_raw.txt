def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    tilt_cos = kwargs["tilt_cos"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    dist_progress = prev_dist - curr_dist
    progress_reward = torch.tanh(2.0 * dist_progress)

    goal_reward = torch.exp(-2.5 * curr_dist)
    reach_bonus = torch.where(curr_dist < 0.5, torch.full_like(curr_dist, 10.0), torch.zeros_like(curr_dist))

    height_error = torch.abs(z_pos - 2.0)
    height_reward = torch.exp(-1.2 * height_error)
    height_penalty = torch.clamp(height_error - 0.25, min=0.0)

    speed_penalty = torch.square(torch.relu(v_norm - 3.5))
    smooth_penalty = 0.05 * action_diff + 0.02 * torch.norm(omega, dim=-1)
    attitude_reward = torch.clamp(tilt_cos, min=-1.0, max=1.0)

    safe_lidar = min_lidar_dist.clamp_min(1e-3)
    obstacle_penalty = 2.0 * torch.exp(-4.0 * safe_lidar) + 0.3 / safe_lidar
    collision_penalty = torch.where(min_lidar_dist < 0.25, torch.full_like(min_lidar_dist, 20.0), torch.zeros_like(min_lidar_dist))

    dir_to_goal = target_pos - curr_pos
    dir_to_goal_norm = torch.norm(dir_to_goal, dim=-1).clamp_min(1e-6)
    dir_to_goal_unit = dir_to_goal / dir_to_goal_norm.unsqueeze(-1)
    alignment = torch.sum(x_body * dir_to_goal_unit, dim=-1)
    alignment_reward = torch.clamp(alignment, min=-1.0, max=1.0)

    reward = (
        4.0 * progress_reward
        + 2.5 * goal_reward
        + reach_bonus
        + 1.0 * height_reward
        - 1.5 * height_penalty
        - 0.3 * speed_penalty
        - smooth_penalty
        + 0.8 * attitude_reward
        + 0.5 * alignment_reward
        - obstacle_penalty
        - collision_penalty
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    return reward
