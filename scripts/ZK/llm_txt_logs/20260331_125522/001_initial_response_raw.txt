def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    tilt_cos = kwargs["tilt_cos"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    eps = 1e-6

    progress = prev_dist - curr_dist
    progress_reward = torch.tanh(2.0 * progress)

    goal_reach = torch.exp(-0.7 * curr_dist) * 8.0
    close_bonus = torch.where(curr_dist < 0.75, 12.0 * (0.75 - curr_dist), torch.zeros_like(curr_dist))

    target_vec = target_pos - curr_pos
    target_dir = target_vec / torch.norm(target_vec, dim=-1, keepdim=True).clamp_min(eps)
    vel_dir = v / v_norm.unsqueeze(-1).clamp_min(eps)
    align_reward = torch.sum(target_dir * vel_dir, dim=-1)
    align_reward = torch.where(v_norm > 0.1, align_reward, torch.zeros_like(align_reward))

    height_error = torch.abs(z_pos - 2.0)
    height_penalty = torch.square(height_error)

    speed_penalty = torch.square(torch.relu(v_norm - 4.0))
    smooth_penalty = torch.square(action_diff) + 0.2 * torch.square(torch.norm(omega, dim=-1))

    upright_reward = tilt_cos.clamp(-1.0, 1.0)

    obstacle_term = torch.exp(-4.0 * min_lidar_dist.clamp_min(0.0))
    obstacle_penalty = 8.0 * obstacle_term + torch.where(min_lidar_dist < 0.5, 6.0 * torch.square(0.5 - min_lidar_dist), torch.zeros_like(min_lidar_dist))
    collision_penalty = torch.where(min_lidar_dist < 0.15, 25.0 * (0.15 - min_lidar_dist).clamp_min(0.0), torch.zeros_like(min_lidar_dist))

    reward = (
        4.0 * progress_reward
        + 1.5 * goal_reach
        + close_bonus
        + 0.8 * align_reward
        + 1.2 * upright_reward
        - 1.8 * height_penalty
        - 0.35 * speed_penalty
        - 0.22 * smooth_penalty
        - 2.2 * obstacle_penalty
        - collision_penalty
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    return reward
