def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    tilt_cos = kwargs["tilt_cos"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    progress = prev_dist - curr_dist
    r_progress = 14.0 * torch.tanh(4.0 * progress)

    r_goal_dist = 8.0 * torch.exp(-0.8 * curr_dist)
    r_success = (curr_dist < 0.7).float() * 18.0 + (curr_dist < 0.35).float() * 28.0

    z_err = torch.abs(z_pos - 2.0)
    r_altitude = -5.0 * torch.tanh(3.0 * torch.relu(z_err - 0.12))
    r_low_alt = -10.0 * torch.relu(1.0 - z_pos)
    r_high_alt = -9.0 * torch.relu(z_pos - 2.08)
    r_ceiling = -(z_pos > 2.12).float() * (5.0 + 14.0 * torch.relu(z_pos - 2.12))
    r_runaway = -14.0 * torch.relu(z_pos - 2.0) * torch.tanh(1.8 * (v_norm + torch.relu(z_pos - 2.0)))

    speed_excess = torch.relu(v_norm - 3.0)
    r_speed = -1.2 * torch.square(speed_excess)
    r_stop = -0.3 * torch.relu(0.08 - v_norm)

    omega_norm = torch.norm(omega, dim=-1)
    r_smooth = -0.22 * torch.tanh(0.7 * action_diff + 0.35 * omega_norm)

    tilt = torch.clamp(tilt_cos, min=-1.0, max=1.0)
    r_upright = 1.4 * torch.relu(tilt - 0.2)

    v_dir = v / v_norm.unsqueeze(-1).clamp_min(1e-6)
    heading_align = (x_body * v_dir).sum(dim=-1)
    r_heading = 0.8 * heading_align * (v_norm > 0.12).float()

    safe_lidar = min_lidar_dist.clamp_min(1e-6)
    r_obstacle_soft = -1.2 * torch.exp(-4.5 * safe_lidar)
    r_obstacle_near = -2.2 * torch.relu(0.8 - safe_lidar).pow(2)
    r_obstacle_hard = -(safe_lidar < 0.35).float() * (2.5 + 6.0 * torch.relu(0.35 - safe_lidar))
    r_collision = -(safe_lidar < 0.12).float() * (6.0 + 12.0 * torch.relu(0.12 - safe_lidar))

    v_z = v[..., 2]
    r_vertical = -0.25 * torch.tanh(torch.abs(v_z) + 0.25 * omega_norm)

    reward = (
        r_progress
        + r_goal_dist
        + r_success
        + r_altitude
        + r_low_alt
        + r_high_alt
        + r_ceiling
        + r_runaway
        + r_speed
        + r_stop
        + r_smooth
        + r_upright
        + r_heading
        + r_obstacle_soft
        + r_obstacle_near
        + r_obstacle_hard
        + r_collision
        + r_vertical
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    return reward
