def compute_reward(**kwargs):
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    tilt_cos = kwargs["tilt_cos"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    dist_delta = prev_dist - curr_dist
    r_progress = 16.0 * torch.tanh(2.5 * dist_delta)
    r_goal_close = 18.0 * torch.exp(-1.2 * curr_dist)
    r_success = (curr_dist < 0.5).float() * 120.0 + (curr_dist < 0.2).float() * 240.0

    z_err = torch.abs(z_pos - 2.0)
    r_alt_hold = -6.0 * torch.square(torch.relu(z_err - 0.10))
    r_alt_soft = -14.0 * torch.square(torch.relu(z_err - 0.25))
    r_low = -20.0 * torch.square(torch.relu(1.0 - z_pos))
    r_high = -14.0 * torch.square(torch.relu(z_pos - 2.02))
    r_ceiling = -(z_pos > 2.03).float() * (18.0 + 22.0 * torch.relu(z_pos - 2.03))
    r_vertical_runaway = -28.0 * torch.relu(z_pos - 2.0) * torch.tanh(1.5 * torch.relu(v_norm - 0.2))

    speed_excess = torch.relu(v_norm - 2.2)
    r_speed = -1.2 * torch.square(speed_excess)
    r_stall = -0.25 * torch.relu(0.08 - v_norm)

    omega_norm = torch.norm(omega, dim=-1)
    r_smooth = -0.03 * torch.tanh(0.8 * action_diff + 0.25 * omega_norm)

    tilt = torch.clamp(tilt_cos, min=-1.0, max=1.0)
    r_upright = 1.0 * torch.relu(tilt - 0.55)

    v_safe = v_norm.clamp_min(1e-6)
    v_dir = v / v_safe.unsqueeze(-1)
    heading_align = (x_body * v_dir).sum(dim=-1)
    r_heading = 0.8 * heading_align * (v_norm > 0.10).float()

    lidar = min_lidar_dist.clamp_min(1e-6)
    r_obstacle_soft = -0.8 * torch.exp(-1.8 * lidar)
    r_obstacle_near = -2.8 * torch.square(torch.relu(0.40 - lidar))
    r_obstacle_hard = -9.0 * torch.square(torch.relu(0.16 - lidar))
    r_collision = -(lidar < 0.08).float() * (8.0 + 18.0 * torch.relu(0.08 - lidar))

    v_z = v[..., 2]
    r_vertical_motion = -0.10 * torch.tanh(torch.abs(v_z) + 0.2 * omega_norm)

    reward = (
        r_progress
        + r_goal_close
        + r_success
        + r_alt_hold
        + r_alt_soft
        + r_low
        + r_high
        + r_ceiling
        + r_vertical_runaway
        + r_speed
        + r_stall
        + r_smooth
        + r_upright
        + r_heading
        + r_obstacle_soft
        + r_obstacle_near
        + r_obstacle_hard
        + r_collision
        + r_vertical_motion
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    return reward
