def compute_reward(**kwargs):
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    tilt_cos = kwargs["tilt_cos"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    progress = prev_dist - curr_dist
    r_progress = 18.0 * torch.tanh(3.0 * progress)

    r_goal_close = 6.0 * torch.exp(-0.9 * curr_dist)
    r_success = (curr_dist < 0.6).float() * 20.0 + (curr_dist < 0.3).float() * 35.0

    z_err = torch.abs(z_pos - 2.0)
    r_altitude = -6.0 * torch.tanh(2.5 * torch.relu(z_err - 0.15))
    r_low_alt = -8.0 * torch.relu(1.05 - z_pos)
    r_high_alt = -10.0 * torch.relu(z_pos - 2.12)
    r_ceiling = -(z_pos > 2.18).float() * (6.0 + 18.0 * torch.relu(z_pos - 2.18))
    r_runaway = -12.0 * torch.relu(z_pos - 2.05) * torch.tanh(1.5 * v_norm + 1.5 * torch.relu(z_pos - 2.05))

    speed_excess = torch.relu(v_norm - 3.2)
    r_speed = -1.0 * torch.square(speed_excess)
    r_stationary = -0.4 * torch.relu(0.12 - v_norm)

    omega_norm = torch.norm(omega, dim=-1)
    r_smooth = -0.25 * torch.tanh(0.8 * action_diff + 0.4 * omega_norm)

    tilt = torch.clamp(tilt_cos, min=-1.0, max=1.0)
    r_upright = 1.2 * tilt

    v_dir = v / v_norm.unsqueeze(-1).clamp_min(1e-6)
    heading_align = (x_body * v_dir).sum(dim=-1)
    r_heading = 0.7 * heading_align * (v_norm > 0.15).float()

    safe_lidar = min_lidar_dist.clamp_min(1e-6)
    r_obstacle_soft = -2.0 * torch.exp(-5.0 * safe_lidar)
    r_obstacle_near = -2.5 * torch.relu(0.7 - safe_lidar).pow(2)
    r_obstacle_hard = -(safe_lidar < 0.35).float() * (3.0 + 8.0 * torch.relu(0.35 - safe_lidar))
    r_collision = -(safe_lidar < 0.15).float() * (8.0 + 15.0 * torch.relu(0.15 - safe_lidar))

    v_z = v[..., 2]
    r_vertical = -0.3 * torch.tanh(torch.abs(v_z) + 0.3 * omega_norm)

    reward = (
        r_progress
        + r_goal_close
        + r_success
        + r_altitude
        + r_low_alt
        + r_high_alt
        + r_ceiling
        + r_runaway
        + r_speed
        + r_stationary
        + r_smooth
        + r_upright
        + r_heading
        + r_obstacle_soft
        + r_obstacle_near
        + r_obstacle_hard
        + r_collision
        + r_vertical
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    return reward
