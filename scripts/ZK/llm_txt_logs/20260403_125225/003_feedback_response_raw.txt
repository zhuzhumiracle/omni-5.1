def compute_reward(**kwargs):
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    tilt_cos = kwargs["tilt_cos"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    progress = prev_dist - curr_dist
    r_progress = 10.0 * torch.tanh(3.0 * progress)

    r_target_close = 18.0 * torch.exp(-0.9 * curr_dist)
    r_success = (curr_dist < 0.6).float() * 20.0 + (curr_dist < 0.3).float() * 35.0

    z_error = torch.abs(z_pos - 2.0)
    r_altitude = -6.0 * torch.tanh(2.5 * torch.relu(z_error - 0.20))
    r_ceiling = -10.0 * torch.relu(z_pos - 2.35)
    r_vertical_runaway = -8.0 * torch.relu(z_pos - 2.15) * torch.sigmoid(1.8 * (v_norm - 0.4))
    r_low_alt = -2.5 * torch.relu(1.05 - z_pos)

    speed_over = torch.relu(v_norm - 3.5)
    r_speed = -0.8 * torch.square(speed_over)
    r_stationary = -0.35 * torch.relu(0.12 - v_norm)

    omega_norm = torch.norm(omega, dim=-1)
    r_smooth = -0.45 * torch.tanh(0.6 * action_diff + 0.35 * omega_norm)

    tilt = torch.clamp(tilt_cos, min=-1.0, max=1.0)
    r_upright = 1.8 * tilt

    v_dir = v / v_norm.unsqueeze(-1).clamp_min(1e-6)
    heading_align = (x_body * v_dir).sum(dim=-1)
    r_heading = 1.0 * heading_align * (v_norm > 0.2).float()

    safe_dist = min_lidar_dist.clamp_min(1e-6)
    r_obstacle = -3.5 * torch.exp(-5.0 * safe_dist) - 1.0 * torch.reciprocal(safe_dist + 0.05)
    r_collision = (min_lidar_dist < 0.35).float() * (-8.0 - 18.0 * torch.relu(0.35 - min_lidar_dist))

    v_z = v[..., 2]
    r_vertical_speed = -0.5 * torch.tanh(torch.abs(v_z) + 0.25 * torch.abs(omega_norm))

    reward = (
        r_progress
        + r_target_close
        + r_success
        + r_altitude
        + r_ceiling
        + r_vertical_runaway
        + r_low_alt
        + r_speed
        + r_stationary
        + r_smooth
        + r_upright
        + r_heading
        + r_obstacle
        + r_collision
        + r_vertical_speed
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    return reward
