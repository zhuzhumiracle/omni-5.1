    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    tilt_cos = kwargs["tilt_cos"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    progress = prev_dist - curr_dist
    progress_reward = torch.tanh(2.0 * progress)

    goal_close = torch.exp(-1.2 * curr_dist)
    goal_bonus = 8.0 * goal_close

    altitude_error = torch.abs(z_pos - 2.0)
    altitude_reward = -1.5 * torch.tanh(altitude_error)

    speed_penalty = torch.square(torch.relu(v_norm - 4.0))
    smooth_penalty = torch.tanh(0.5 * action_diff) + 0.25 * torch.tanh(torch.norm(omega, dim=-1))

    safe_dist = min_lidar_dist.clamp_min(1e-3)
    obstacle_penalty = -2.0 * torch.exp(-3.5 * safe_dist) - 0.8 * torch.reciprocal(safe_dist + 0.05)

        - 0.8 * smooth_penalty
        + obstacle_penalty
        + collision_penalty
    )
