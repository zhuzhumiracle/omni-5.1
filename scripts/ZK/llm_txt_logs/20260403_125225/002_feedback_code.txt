def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    tilt_cos = kwargs["tilt_cos"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    progress = prev_dist - curr_dist
    progress_reward = 8.0 * torch.tanh(2.5 * progress)

    target_proximity = torch.exp(-0.8 * curr_dist)
    target_reward = 12.0 * target_proximity

    success_bonus = (curr_dist < 0.6).float() * 25.0 + (curr_dist < 0.3).float() * 40.0

    altitude_error = torch.abs(z_pos - 2.0)
    altitude_penalty = -3.0 * torch.tanh(1.5 * torch.relu(altitude_error - 0.25))
    ceiling_penalty = -4.0 * torch.relu(z_pos - 2.6)
    runaway_penalty = -3.5 * torch.relu(z_pos - 2.2) * torch.sigmoid(2.0 * (v_norm - 0.5))

    speed_penalty = -0.6 * torch.square(torch.relu(v_norm - 4.0))
    stop_penalty = -0.4 * torch.relu(0.15 - v_norm)

    omega_norm = torch.norm(omega, dim=-1)
    smooth_penalty = -0.5 * torch.tanh(0.5 * action_diff + 0.25 * omega_norm)

    upright_reward = 1.5 * torch.clamp(tilt_cos, min=-1.0, max=1.0)

    v_dir = v / v_norm.unsqueeze(-1).clamp_min(1e-6)
    heading_align = (x_body * v_dir).sum(dim=-1)
    heading_reward = 0.8 * heading_align * (v_norm > 0.2).float()

    safe_dist = min_lidar_dist.clamp_min(1e-6)
    obstacle_penalty = -4.0 * torch.exp(-6.0 * safe_dist) - 1.5 * torch.reciprocal(safe_dist + 0.02)
    collision_penalty = (min_lidar_dist < 0.35).float() * (-3.0 - 10.0 * torch.relu(0.35 - min_lidar_dist))

    stability_penalty = -0.3 * torch.tanh(torch.abs(v[... , 2]) + 0.5 * torch.abs(omega_norm))

    reward = (
        progress_reward
        + target_reward
        + success_bonus
        + altitude_penalty
        + ceiling_penalty
        + runaway_penalty
        + speed_penalty
        + stop_penalty
        + smooth_penalty
        + upright_reward
        + heading_reward
        + obstacle_penalty
        + collision_penalty
        + stability_penalty
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    return reward
