def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    tilt_cos = kwargs["tilt_cos"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    progress = prev_dist - curr_dist
    progress_reward = torch.tanh(2.0 * progress)

    goal_close = torch.exp(-1.2 * curr_dist)
    goal_bonus = 8.0 * goal_close

    target_reach_bonus = (curr_dist < 0.6).float() * 20.0 + (curr_dist < 0.3).float() * 35.0

    altitude_error = torch.abs(z_pos - 2.0)
    altitude_reward = -1.5 * torch.tanh(altitude_error)

    speed_penalty = torch.square(torch.relu(v_norm - 4.0))
    smooth_penalty = torch.tanh(0.5 * action_diff) + 0.25 * torch.tanh(torch.norm(omega, dim=-1))

    upright_reward = torch.clamp(tilt_cos, min=-1.0, max=1.0)

    v_dir = v / v_norm.unsqueeze(-1).clamp_min(1e-6)
    align = (x_body * v_dir).sum(dim=-1)
    align_reward = align * (v_norm > 0.25).float()

    safe_dist = min_lidar_dist.clamp_min(1e-3)
    obstacle_penalty = -2.0 * torch.exp(-3.5 * safe_dist) - 0.8 * torch.reciprocal(safe_dist + 0.05)

    collision_penalty = (min_lidar_dist < 0.35).float() * (-4.0 - 8.0 * torch.relu(0.35 - min_lidar_dist))

    reward = (
        6.0 * progress_reward
        + goal_bonus
        + target_reach_bonus
        + altitude_reward
        + 1.5 * upright_reward
        + 0.5 * align_reward
        - 0.35 * speed_penalty
        - 0.8 * smooth_penalty
        + obstacle_penalty
        + collision_penalty
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    return reward
