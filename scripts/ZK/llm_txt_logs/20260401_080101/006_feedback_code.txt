def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    progress = prev_dist - curr_dist
    r_progress = torch.tanh(6.0 * progress)

    dist_to_goal = curr_dist.clamp_min(0.0)
    r_goal_close = torch.exp(-0.35 * dist_to_goal)
    success = (curr_dist < 0.8).to(curr_dist.dtype)
    r_success = success * 3000.0

    direction = target_pos - curr_pos
    direction_norm = torch.norm(direction, dim=-1).clamp_min(1e-6)
    direction_unit = direction / direction_norm.unsqueeze(-1)
    align = (x_body * direction_unit).sum(dim=-1).clamp(-1.0, 1.0)
    r_align = torch.relu(align)

    v_align = torch.norm(v, dim=-1)
    r_move = torch.tanh(v_align / 3.0) * torch.relu(align)

    z_err = torch.abs(z_pos - 2.0)
    r_z_pen = torch.relu(z_err - 0.25)

    omega_norm = torch.norm(omega, dim=-1)
    r_omega = torch.tanh(omega_norm / 4.0)

    r_smooth = torch.tanh(action_diff / 3.0)

    speed_pen = torch.relu(v_norm - 5.0).pow(2)

    lidar = min_lidar_dist.clamp_min(1e-6)
    inv_lidar = 1.0 / lidar
    obstacle_hard = -20.0 * inv_lidar - 80.0 * inv_lidar.pow(2) - 300.0 * inv_lidar.pow(3)
    obstacle_soft = -8.0 * torch.exp(-5.0 * min_lidar_dist.clamp_min(0.0))
    danger = (min_lidar_dist < 1.0).to(min_lidar_dist.dtype) * (-120.0 * torch.exp(-3.0 * min_lidar_dist.clamp_min(0.0)))
    collision = (min_lidar_dist < 0.35).to(min_lidar_dist.dtype) * (-5000.0 - 20000.0 * torch.relu(0.35 - min_lidar_dist))

    reward = (
        60.0 * r_progress
        + 18.0 * r_goal_close
        + 6.0 * r_align
        + 2.0 * r_move
        - 4.0 * r_z_pen
        - 0.4 * r_omega
        - 0.2 * r_smooth
        - 0.8 * speed_pen
        + obstacle_hard
        + obstacle_soft
        + danger
        + collision
        + r_success
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    return reward
