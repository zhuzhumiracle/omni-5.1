def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    progress = prev_dist - curr_dist
    r_progress = torch.tanh(10.0 * progress)

    dist_term = torch.exp(-0.8 * curr_dist)
    near_goal_term = torch.exp(-0.35 * curr_dist)
    success_bonus = (curr_dist < 0.8).to(curr_dist.dtype) * 150.0

    direction = target_pos - curr_pos
    direction_norm = torch.norm(direction, dim=-1).clamp_min(1e-6)
    direction_unit = direction / direction_norm.unsqueeze(-1)
    align = (x_body * direction_unit).sum(dim=-1).clamp(-1.0, 1.0)
    r_align = align * torch.tanh(v_norm / 2.5)

    z_err = torch.abs(z_pos - 2.0)
    r_z = torch.relu(z_err - 0.25)

    omega_norm = torch.norm(omega, dim=-1)
    r_omega = torch.tanh(omega_norm / 6.0)

    r_smooth = torch.tanh(action_diff / 5.0)

    speed_penalty = torch.relu(v_norm - 5.0).pow(2)

    lidar = min_lidar_dist.clamp_min(1e-6)
    obstacle_penalty = -0.8 / lidar - 2.5 / lidar.pow(2) - 8.0 / lidar.pow(3)
    danger_penalty = -30.0 * torch.exp(-10.0 * min_lidar_dist.clamp_min(0.0))
    collision_penalty = (min_lidar_dist < 0.35).to(min_lidar_dist.dtype) * (-300.0 - 800.0 * torch.relu(0.35 - min_lidar_dist))

    reward = (
        15.0 * r_progress
        + 8.0 * dist_term
        + 4.0 * near_goal_term
        + success_bonus
        + 2.5 * r_align
        - 4.0 * r_z
        - 0.4 * r_omega
        - 0.2 * r_smooth
        - 0.3 * speed_penalty
        + obstacle_penalty
        + danger_penalty
        + collision_penalty
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    return reward
