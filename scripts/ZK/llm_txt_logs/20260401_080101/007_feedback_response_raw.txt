def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    eps = 1e-6

    progress = prev_dist - curr_dist
    r_progress = torch.tanh(8.0 * progress)

    dist_term = torch.exp(-0.25 * curr_dist.clamp_min(0.0))
    close_bonus = torch.exp(-0.8 * curr_dist.clamp_min(0.0))

    success = (curr_dist < 0.8).to(curr_dist.dtype)
    r_success = success * 5000.0

    direction = target_pos - curr_pos
    direction_norm = torch.norm(direction, dim=-1).clamp_min(eps)
    direction_unit = direction / direction_norm.unsqueeze(-1)
    align = (x_body * direction_unit).sum(dim=-1).clamp(-1.0, 1.0)
    r_align = torch.relu(align)

    speed = torch.norm(v, dim=-1)
    forward_speed = speed * torch.relu(align)
    r_forward_motion = torch.tanh(forward_speed / 4.0)

    z_err = torch.abs(z_pos - 2.0)
    r_z = torch.relu(z_err - 0.25)

    omega_norm = torch.norm(omega, dim=-1)
    r_omega = torch.tanh(omega_norm / 3.0)

    r_smooth = torch.tanh(action_diff / 2.0)

    speed_pen = torch.relu(v_norm - 5.0).pow(2)

    lidar = min_lidar_dist.clamp_min(eps)
    inv_lidar = 1.0 / lidar
    collision_zone = (min_lidar_dist < 0.8).to(min_lidar_dist.dtype)
    near_zone = (min_lidar_dist < 2.0).to(min_lidar_dist.dtype)

    obstacle_severe = -200.0 * inv_lidar - 2000.0 * inv_lidar.pow(2) - 10000.0 * inv_lidar.pow(3)
    obstacle_bump = -60.0 * torch.exp(-6.0 * min_lidar_dist.clamp_min(0.0))
    collision_penalty = collision_zone * (-20000.0 - 50000.0 * torch.relu(0.5 - min_lidar_dist))
    near_penalty = near_zone * (-200.0 * torch.exp(-2.0 * min_lidar_dist.clamp_min(0.0)))

    reward = (
        40.0 * r_progress
        + 12.0 * dist_term
        + 8.0 * close_bonus
        + 4.0 * r_align
        + 3.0 * r_forward_motion
        - 6.0 * r_z
        - 0.8 * r_omega
        - 0.3 * r_smooth
        - 1.2 * speed_pen
        + obstacle_severe
        + obstacle_bump
        + near_penalty
        + collision_penalty
        + r_success
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    return reward
