def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    tilt_cos = kwargs["tilt_cos"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    progress = prev_dist - curr_dist
    progress_reward = torch.tanh(5.0 * progress)

    dist_reward = torch.exp(-1.5 * curr_dist)
    near_goal_bonus = (curr_dist < 2.0).to(curr_dist.dtype) * (2.0 * torch.exp(-0.8 * curr_dist))
    success_bonus = (curr_dist < 0.5).to(curr_dist.dtype) * 50.0

    direction_vec = target_pos - curr_pos
    direction_norm = torch.norm(direction_vec, dim=-1).clamp_min(1e-6)
    direction_unit = direction_vec / direction_norm.unsqueeze(-1)
    forward_align = (x_body * direction_unit).sum(dim=-1)
    align_reward = torch.clamp(forward_align, min=-1.0, max=1.0) * torch.tanh(v_norm / 4.0)

    z_error = torch.abs(z_pos - 2.0)
    z_penalty = torch.relu(z_error - 0.4)
    upright_penalty = torch.relu(1.0 - tilt_cos)

    speed_penalty = torch.relu(v_norm - 5.0).pow(2)
    omega_penalty = torch.norm(omega, dim=-1).clamp_max(20.0)
    smooth_penalty = torch.clamp(action_diff, max=20.0)

    safe_dist = min_lidar_dist.clamp_min(1e-6)
    collision_barrier = 8.0 / safe_dist + 4.0 / safe_dist.pow(2)
    danger_penalty = 6.0 * torch.exp(-3.0 * min_lidar_dist.clamp_min(0.0))
    hard_collision_penalty = (min_lidar_dist < 0.25).to(min_lidar_dist.dtype) * (100.0 + 200.0 * torch.relu(0.25 - min_lidar_dist))

    reward = (
        8.0 * progress_reward
        + 4.0 * dist_reward
        + near_goal_bonus
        + success_bonus
        + 1.5 * align_reward
        - 2.0 * z_penalty
        - 1.5 * upright_penalty
        - 0.3 * speed_penalty
        - 0.1 * omega_penalty
        - 0.15 * smooth_penalty
        - 3.0 * danger_penalty
        - 0.5 * collision_barrier
        - hard_collision_penalty
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    return reward
