def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    eps = 1e-6

    dist_delta = prev_dist - curr_dist
    r_progress = torch.tanh(6.0 * dist_delta)

    r_dist = torch.exp(-0.35 * curr_dist.clamp_min(0.0))
    r_close = torch.exp(-0.9 * curr_dist.clamp_min(0.0))

    success = (curr_dist < 0.8).to(curr_dist.dtype)
    r_success = success * 10000.0

    direction = target_pos - curr_pos
    direction_norm = torch.norm(direction, dim=-1).clamp_min(eps)
    direction_unit = direction / direction_norm.unsqueeze(-1)
    align = (x_body * direction_unit).sum(dim=-1).clamp(-1.0, 1.0)
    r_align = torch.relu(align)

    speed = torch.norm(v, dim=-1)
    r_forward = torch.tanh(speed * torch.relu(align) / 4.0)

    z_err = torch.abs(z_pos - 2.0)
    r_z = torch.relu(z_err - 0.25)

    omega_norm = torch.norm(omega, dim=-1)
    r_omega = torch.tanh(omega_norm / 3.0)

    r_smooth = torch.tanh(action_diff / 1.5)

    speed_pen = torch.relu(v_norm - 5.0).pow(2)

    d = min_lidar_dist.clamp_min(eps)
    inv_d = 1.0 / d

    collision = (min_lidar_dist < 0.5).to(min_lidar_dist.dtype)
    near = (min_lidar_dist < 1.5).to(min_lidar_dist.dtype)

    obstacle_hard = -5000.0 * inv_d.pow(2) - 20000.0 * inv_d.pow(3)
    obstacle_soft = -80.0 * torch.exp(-4.0 * min_lidar_dist.clamp_min(0.0))
    obstacle_near = -300.0 * torch.exp(-2.5 * min_lidar_dist.clamp_min(0.0))
    obstacle_collision = collision * (-50000.0 - 50000.0 * torch.relu(0.3 - min_lidar_dist))
    obstacle_close = near * (-1000.0 * torch.exp(-3.0 * min_lidar_dist.clamp_min(0.0)))

    reward = (
        80.0 * r_progress
        + 18.0 * r_dist
        + 10.0 * r_close
        + 6.0 * r_align
        + 4.0 * r_forward
        - 8.0 * r_z
        - 1.0 * r_omega
        - 0.5 * r_smooth
        - 1.5 * speed_pen
        + obstacle_hard
        + obstacle_soft
        + obstacle_near
        + obstacle_close
        + obstacle_collision
        + r_success
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    return reward
