def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    tilt_cos = kwargs["tilt_cos"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    progress = prev_dist - curr_dist
    progress_reward = torch.tanh(3.0 * progress)

    target_bonus = torch.exp(-2.5 * curr_dist) * 4.0
    close_reach_bonus = (curr_dist < 0.75).to(curr_dist.dtype) * (12.0 * torch.exp(-2.0 * curr_dist))

    z_error = z_pos - 2.0
    z_penalty = torch.square(torch.clamp(torch.abs(z_error) - 0.25, min=0.0))

    speed_penalty = torch.square(torch.relu(v_norm - 4.0)) * 0.25
    smooth_penalty = 0.08 * action_diff + 0.03 * torch.norm(omega, dim=-1)

    upright_bonus = torch.clamp(tilt_cos, min=0.0, max=1.0)

    direction_vec = target_pos - curr_pos
    direction_norm = torch.norm(direction_vec, dim=-1).clamp_min(1e-6)
    direction_unit = direction_vec / direction_norm.unsqueeze(-1)
    forward_align = (x_body * direction_unit).sum(dim=-1)
    align_bonus = torch.clamp(forward_align, min=-1.0, max=1.0) * torch.clamp(v_norm, max=8.0) / 8.0

    obstacle_penalty = 3.0 * torch.exp(-4.0 * min_lidar_dist.clamp_min(0.0)) + 8.0 * torch.relu(0.6 - min_lidar_dist).pow(2)
    severe_collision_penalty = (min_lidar_dist < 0.15).to(min_lidar_dist.dtype) * 25.0

    reward = (
        3.5 * progress_reward
        + target_bonus
        + close_reach_bonus
        + 1.5 * upright_bonus
        + 0.8 * align_bonus
        - 1.2 * z_penalty
        - speed_penalty
        - smooth_penalty
        - obstacle_penalty
        - severe_collision_penalty
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    return reward
