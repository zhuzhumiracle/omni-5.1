def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    eps = 1e-6

    dist_delta = prev_dist - curr_dist
    r_progress = torch.tanh(10.0 * dist_delta)
    r_near_goal = torch.exp(-0.7 * curr_dist.clamp_min(0.0))

    goal_reached = (curr_dist < 0.75).to(curr_dist.dtype)
    r_goal = goal_reached * 20000.0

    direction = target_pos - curr_pos
    direction_norm = torch.norm(direction, dim=-1).clamp_min(eps)
    direction_unit = direction / direction_norm.unsqueeze(-1)
    align = (x_body * direction_unit).sum(dim=-1).clamp(-1.0, 1.0)
    r_align = torch.relu(align)

    speed = torch.norm(v, dim=-1)
    r_forward = torch.tanh((speed * torch.relu(align)) / 2.5)

    z_err = torch.abs(z_pos - 2.0)
    r_alt = torch.relu(z_err - 0.25)

    omega_norm = torch.norm(omega, dim=-1)
    r_omega = torch.tanh(omega_norm / 2.0)

    r_smooth = torch.tanh(action_diff / 0.5)

    speed_pen = torch.relu(v_norm - 5.0).pow(2)

    d = min_lidar_dist.clamp_min(eps)
    inv_d = 1.0 / d
    inv_d2 = inv_d * inv_d
    inv_d4 = inv_d2 * inv_d2

    collision = (min_lidar_dist < 0.5).to(min_lidar_dist.dtype)
    very_close = (min_lidar_dist < 1.0).to(min_lidar_dist.dtype)
    near = (min_lidar_dist < 2.0).to(min_lidar_dist.dtype)

    obstacle_hard = -2000.0 * inv_d2 - 20000.0 * inv_d4
    obstacle_soft = -100.0 * torch.exp(-2.5 * min_lidar_dist.clamp_min(0.0))
    obstacle_near = -300.0 * torch.exp(-1.5 * min_lidar_dist.clamp_min(0.0))
    obstacle_close = -1000.0 * torch.exp(-3.0 * min_lidar_dist.clamp_min(0.0)) * very_close
    obstacle_collision = collision * (-50000.0 - 500000.0 * torch.relu(0.5 - min_lidar_dist) / 0.5)
    obstacle_near_mask = near * obstacle_soft

    reward = (
        80.0 * r_progress
        + 12.0 * r_near_goal
        + 6.0 * r_align
        + 4.0 * r_forward
        - 10.0 * r_alt
        - 2.5 * r_omega
        - 1.0 * r_smooth
        - 3.0 * speed_pen
        + obstacle_hard
        + obstacle_near
        + obstacle_close
        + obstacle_near_mask
        + obstacle_collision
        + r_goal
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    return reward
