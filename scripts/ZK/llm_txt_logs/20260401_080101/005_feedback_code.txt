def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    progress = prev_dist - curr_dist
    r_progress = torch.tanh(8.0 * progress)

    goal_close = torch.exp(-0.25 * curr_dist)
    goal_bonus = (curr_dist < 0.75).to(curr_dist.dtype) * 400.0
    near_goal_bonus = (curr_dist < 3.0).to(curr_dist.dtype) * 25.0 * torch.exp(-0.8 * curr_dist)

    direction = target_pos - curr_pos
    direction_norm = torch.norm(direction, dim=-1).clamp_min(1e-6)
    direction_unit = direction / direction_norm.unsqueeze(-1)
    align = (x_body * direction_unit).sum(dim=-1).clamp(-1.0, 1.0)
    r_align = torch.relu(align) * torch.tanh(v_norm / 2.0)

    z_err = torch.abs(z_pos - 2.0)
    r_z = torch.relu(z_err - 0.25)

    omega_norm = torch.norm(omega, dim=-1)
    r_omega = torch.tanh(omega_norm / 5.0)

    r_smooth = torch.tanh(action_diff / 4.0)

    speed_penalty = torch.relu(v_norm - 5.0).pow(2)

    lidar = min_lidar_dist.clamp_min(1e-6)
    obstacle_barrier = -2.0 / lidar - 8.0 / lidar.pow(2) - 30.0 / lidar.pow(3)
    obstacle_near = -50.0 * torch.exp(-8.0 * min_lidar_dist.clamp_min(0.0))
    collision_penalty = (min_lidar_dist < 0.35).to(min_lidar_dist.dtype) * (-1000.0 - 4000.0 * torch.relu(0.35 - min_lidar_dist))
    danger_penalty = (min_lidar_dist < 0.8).to(min_lidar_dist.dtype) * (-100.0 * torch.exp(-4.0 * min_lidar_dist.clamp_min(0.0)))

    reward = (
        40.0 * r_progress
        + 12.0 * goal_close
        + near_goal_bonus
        + goal_bonus
        + 4.0 * r_align
        - 3.0 * r_z
        - 0.2 * r_omega
        - 0.1 * r_smooth
        - 0.4 * speed_penalty
        + obstacle_barrier
        + obstacle_near
        + danger_penalty
        + collision_penalty
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    return reward
