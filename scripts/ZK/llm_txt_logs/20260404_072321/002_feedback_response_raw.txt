def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    tilt_cos = kwargs["tilt_cos"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    eps = 1e-6
    target_altitude = 2.0

    target_vec = target_pos - curr_pos
    target_dir = target_vec / torch.norm(target_vec, dim=-1).clamp_min(eps).unsqueeze(-1)
    forward_align = (x_body * target_dir).sum(dim=-1).clamp(-1.0, 1.0)

    omega_norm = torch.norm(omega, dim=-1)
    vz_abs = torch.abs(v[:, 2])

    progress_reward = torch.tanh(4.0 * (prev_dist - curr_dist))
    proximity_reward = 2.2 * torch.exp(-0.28 * curr_dist)
    forward_reward = 0.7 * torch.relu(forward_align) * torch.exp(-0.12 * curr_dist)
    goal_bonus = (curr_dist < 0.75).float() * 16.0

    safe_dist = min_lidar_dist.clamp_min(eps)
    near_mask = (min_lidar_dist < 2.0).float()
    critical_mask = (min_lidar_dist < 0.5).float()

    obstacle_penalty = (
        0.8 * torch.exp(-1.2 * safe_dist)
        + 2.6 * torch.exp(-3.8 * safe_dist)
        + 4.0 * torch.exp(-9.0 * safe_dist)
        + critical_mask * 6.0
    )

    near_speed_penalty = near_mask * (0.35 * v_norm + 0.12 * torch.square(v_norm))
    near_vz_penalty = near_mask * (0.55 * vz_abs + 0.15 * torch.square(vz_abs))
    near_omega_penalty = near_mask * (0.18 * omega_norm + 0.05 * torch.square(omega_norm))

    low_speed_penalty = 0.25 * torch.relu(0.9 - v_norm)
    slow_near_obstacle_penalty = near_mask * 0.35 * torch.relu(1.0 - v_norm)
    upward_escape_penalty = near_mask * 0.45 * torch.relu(v[:, 2])

    smooth_penalty = 0.10 * action_diff + 0.08 * omega_norm + 0.02 * torch.square(action_diff)
    global_tilt_penalty = 0.9 * torch.relu(0.88 - tilt_cos) + 0.18 * torch.square(torch.relu(0.96 - tilt_cos))

    altitude_reg = -0.18 * torch.square(z_pos - target_altitude)

    is_flipped = tilt_cos < 0.18
    is_out_of_bounds = (z_pos < 0.8) | (z_pos > 3.5)
    death_penalty = (is_flipped | is_out_of_bounds).float() * 40.0

    reward = (
        5.0 * progress_reward
        + 2.0 * proximity_reward
        + forward_reward
        + goal_bonus
        + altitude_reg
        - obstacle_penalty
        - near_speed_penalty
        - near_vz_penalty
        - near_omega_penalty
        - low_speed_penalty
        - slow_near_obstacle_penalty
        - upward_escape_penalty
        - smooth_penalty
        - global_tilt_penalty
        - death_penalty
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    return reward
