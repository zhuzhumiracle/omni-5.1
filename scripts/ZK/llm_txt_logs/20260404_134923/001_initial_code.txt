def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    tilt_cos = kwargs["tilt_cos"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    eps = 1e-6

    target_altitude = 2.0

    target_vec = target_pos - curr_pos
    target_dist_safe = torch.norm(target_vec, dim=-1).clamp_min(eps)
    target_dir = target_vec / target_dist_safe.unsqueeze(-1)
    forward_align = (x_body * target_dir).sum(dim=-1).clamp(-1.0, 1.0)

    progress = prev_dist - curr_dist
    progress_reward = torch.tanh(2.5 * progress)
    dist_reward = torch.exp(-0.25 * curr_dist)

    goal_radius = 6.0
    goal_bonus = (curr_dist < goal_radius).float() * 8.0
    close_goal_bonus = torch.exp(-1.2 * curr_dist) * 3.0

    speed_target = 2.0
    speed_low_penalty = torch.relu(speed_target - v_norm)
    speed_high_penalty = torch.relu(v_norm - speed_target)
    speed_band_penalty = 0.35 * torch.square(speed_low_penalty) + 0.25 * torch.square(speed_high_penalty)

    omega_norm = torch.norm(omega, dim=-1)
    smooth_penalty = 0.08 * action_diff + 0.04 * omega_norm + 0.03 * torch.square(omega_norm)

    global_tilt_penalty = 0.8 * torch.relu(0.85 - tilt_cos) + 0.15 * torch.square(torch.relu(0.95 - tilt_cos))

    altitude_reg = -0.35 * torch.square(z_pos - target_altitude)

    flip_mask = tilt_cos < 0.2
    out_z_mask = (z_pos < 0.8) | (z_pos > 3.5)
    death_penalty = (flip_mask | out_z_mask).float() * 60.0

    safe_dist = min_lidar_dist.clamp_min(eps)
    near_mask = (min_lidar_dist < 2.0).float()
    critical_mask = (min_lidar_dist < 0.6).float()

    obstacle_penalty = (
        0.35 * torch.exp(-1.2 * safe_dist)
        + 1.2 * torch.exp(-4.5 * safe_dist)
        + 4.0 * torch.relu(0.4 - safe_dist)
        + 10.0 * torch.relu(0.2 - safe_dist)
    )

    vz = v[:, 2]
    near_speed_penalty = near_mask * (0.18 * v_norm + 0.08 * torch.square(v_norm))
    near_vz_penalty = near_mask * (0.25 * torch.abs(vz) + 0.10 * torch.square(vz))
    near_omega_penalty = near_mask * (0.10 * omega_norm + 0.05 * torch.square(omega_norm))
    near_upward_escape_penalty = near_mask * torch.relu(vz) * 0.25

    collision_risk_penalty = critical_mask * (
        0.40 * v_norm
        + 0.20 * torch.abs(vz)
        + 0.12 * omega_norm
        + 0.08 * torch.square(v_norm)
        + 0.08 * torch.square(torch.abs(vz))
        + 0.05 * torch.square(omega_norm)
    )

    alignment_reward = 0.8 * torch.relu(forward_align) * torch.exp(-0.15 * curr_dist)

    reward = (
        3.5 * progress_reward
        + 1.4 * dist_reward
        + 0.6 * close_goal_bonus
        + goal_bonus
        + alignment_reward
        + altitude_reg
        - obstacle_penalty
        - near_speed_penalty
        - near_vz_penalty
        - near_omega_penalty
        - near_upward_escape_penalty
        - collision_risk_penalty
        - speed_band_penalty
        - smooth_penalty
        - global_tilt_penalty
        - death_penalty
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    return reward
