def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    z_pos = kwargs["z_pos"]
    tilt_cos = kwargs["tilt_cos"]
    x_body = kwargs["x_body"]

    eps = 1e-6
    target_altitude = 2.0
    goal_radius = 6.0

    target_vec = target_pos - curr_pos
    target_dir = target_vec / torch.norm(target_vec, dim=-1).clamp_min(eps).unsqueeze(-1)
    forward_align = (x_body * target_dir).sum(dim=-1).clamp(-1.0, 1.0)

    progress = prev_dist - curr_dist
    progress_reward = torch.tanh(2.8 * progress) * torch.tanh(0.6 * torch.relu(progress) + 0.4)
    proximity_reward = 1.25 * torch.exp(-0.22 * curr_dist)
    goal_bonus = (curr_dist < goal_radius).float() * 2.5
    close_goal_bonus = torch.exp(-0.8 * curr_dist) * 1.2
    forward_reward = 0.55 * torch.relu(forward_align) * torch.exp(-0.06 * curr_dist)

    omega_norm = torch.norm(omega, dim=-1)
    smooth_penalty = 0.08 * action_diff + 0.05 * omega_norm + 0.03 * torch.square(omega_norm)

    global_tilt_penalty = 0.65 * torch.relu(0.90 - tilt_cos) + 0.18 * torch.square(torch.relu(0.95 - tilt_cos))

    altitude_reg = -0.18 * torch.square(z_pos - target_altitude)

    is_flipped = tilt_cos < 0.18
    is_out_of_bounds = (z_pos < 0.8) | (z_pos > 3.5)
    death_penalty = (is_flipped | is_out_of_bounds).float() * 40.0

    safe_dist = min_lidar_dist.clamp_min(eps)
    near_mask = (min_lidar_dist < 2.0).float()
    critical_mask = (min_lidar_dist < 0.4).float()

    obstacle_penalty = (
        0.25 * torch.exp(-1.2 * safe_dist)
        + 0.85 * torch.exp(-3.6 * safe_dist)
        + 1.8 * torch.relu(0.60 - safe_dist)
        + 3.5 * torch.relu(0.30 - safe_dist)
        + 5.5 * critical_mask
    )

    vz = v[:, 2]
    near_speed_penalty = near_mask * (0.22 * v_norm + 0.10 * torch.square(v_norm))
    near_vz_penalty = near_mask * (0.28 * torch.abs(vz) + 0.14 * torch.square(vz))
    near_omega_penalty = near_mask * (0.12 * omega_norm + 0.05 * torch.square(omega_norm))
    near_upward_escape_penalty = near_mask * torch.relu(vz) * 0.35

    low_speed_penalty = 0.30 * torch.relu(1.0 - v_norm) + 0.14 * torch.square(torch.relu(1.0 - v_norm))
    high_speed_penalty = 0.18 * torch.relu(v_norm - 2.7) + 0.08 * torch.square(torch.relu(v_norm - 2.7))
    speed_band_penalty = low_speed_penalty + high_speed_penalty

    collision_risk_penalty = critical_mask * (
        0.25 * v_norm
        + 0.18 * torch.abs(vz)
        + 0.14 * omega_norm
        + 0.05 * torch.square(v_norm)
        + 0.05 * torch.square(torch.abs(vz))
        + 0.04 * torch.square(omega_norm)
    )

    reward = (
        4.5 * progress_reward
        + 1.6 * proximity_reward
        + 0.9 * close_goal_bonus
        + goal_bonus
        + forward_reward
        + altitude_reg
        - obstacle_penalty
        - near_speed_penalty
        - near_vz_penalty
        - near_omega_penalty
        - near_upward_escape_penalty
        - collision_risk_penalty
        - speed_band_penalty
        - smooth_penalty
        - global_tilt_penalty
        - death_penalty
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    return reward
