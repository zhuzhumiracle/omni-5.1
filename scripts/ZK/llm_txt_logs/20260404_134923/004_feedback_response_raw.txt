def compute_reward(**kwargs):
    curr_pos = kwargs["curr_pos"]
    target_pos = kwargs["target_pos"]
    curr_dist = kwargs["curr_dist"]
    prev_dist = kwargs["prev_dist"]
    v = kwargs["v"]
    v_norm = kwargs["v_norm"]
    omega = kwargs["omega"]
    min_lidar_dist = kwargs["min_lidar_dist"]
    action_diff = kwargs["action_diff"]
    tilt_cos = kwargs["tilt_cos"]
    z_pos = kwargs["z_pos"]
    x_body = kwargs["x_body"]

    eps = 1e-6
    target_altitude = 2.0

    target_vec = target_pos - curr_pos
    target_dist_safe = torch.norm(target_vec, dim=-1).clamp_min(eps)
    target_dir = target_vec / target_dist_safe.unsqueeze(-1)
    forward_align = (x_body * target_dir).sum(dim=-1).clamp(-1.0, 1.0)

    progress = prev_dist - curr_dist
    progress_reward = torch.tanh(2.2 * progress)
    proximity_reward = torch.exp(-0.22 * curr_dist)
    goal_bonus = (curr_dist < 6.0).float() * 10.0
    close_goal_bonus = torch.exp(-0.9 * curr_dist) * 1.5

    forward_reward = 0.6 * torch.relu(forward_align) * torch.exp(-0.10 * curr_dist)

    omega_norm = torch.norm(omega, dim=-1)
    smooth_penalty = 0.06 * action_diff + 0.05 * omega_norm + 0.02 * torch.square(omega_norm)

    global_tilt_penalty = 0.7 * torch.relu(0.88 - tilt_cos) + 0.12 * torch.square(torch.relu(0.95 - tilt_cos))

    altitude_reg = -0.18 * torch.square(z_pos - target_altitude)

    is_flipped = tilt_cos < 0.18
    is_out_of_bounds = (z_pos < 0.8) | (z_pos > 3.5)
    death_penalty = (is_flipped | is_out_of_bounds).float() * 45.0

    safe_dist = min_lidar_dist.clamp_min(eps)
    near_mask = (min_lidar_dist < 2.0).float()
    critical_mask = (min_lidar_dist < 0.4).float()

    obstacle_penalty = (
        0.25 * torch.exp(-1.0 * safe_dist)
        + 0.90 * torch.exp(-3.2 * safe_dist)
        + 2.50 * torch.exp(-8.0 * safe_dist)
        + 1.80 * torch.relu(0.45 - safe_dist)
        + 4.50 * torch.relu(0.22 - safe_dist)
    )

    vz = v[:, 2]
    near_speed_penalty = near_mask * (
        0.16 * v_norm
        + 0.08 * torch.square(v_norm)
        + 0.05 * torch.relu(1.0 - v_norm)
        + 0.04 * torch.relu(v_norm - 3.2)
    )
    near_vz_penalty = near_mask * (
        0.22 * torch.abs(vz)
        + 0.10 * torch.square(vz)
        + 0.12 * torch.relu(vz)
    )
    near_omega_penalty = near_mask * (
        0.10 * omega_norm
        + 0.04 * torch.square(omega_norm)
    )
    near_upward_escape_penalty = near_mask * (0.22 * torch.relu(vz) + 0.06 * torch.square(torch.relu(vz)))

    collision_risk_penalty = critical_mask * (
        0.28 * v_norm
        + 0.18 * torch.abs(vz)
        + 0.12 * omega_norm
        + 0.10 * torch.square(v_norm)
        + 0.08 * torch.square(torch.abs(vz))
        + 0.05 * torch.square(omega_norm)
    )

    reward = (
        4.2 * progress_reward
        + 1.8 * proximity_reward
        + 0.8 * close_goal_bonus
        + goal_bonus
        + forward_reward
        + altitude_reg
        - obstacle_penalty
        - near_speed_penalty
        - near_vz_penalty
        - near_omega_penalty
        - near_upward_escape_penalty
        - collision_risk_penalty
        - smooth_penalty
        - global_tilt_penalty
        - death_penalty
    )

    reward = torch.nan_to_num(reward, nan=-1e4, posinf=1e4, neginf=-1e4)
    reward = torch.clamp(reward, min=-1e4, max=1e4)
    return reward
